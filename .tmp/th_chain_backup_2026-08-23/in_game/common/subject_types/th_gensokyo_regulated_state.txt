### ============================================================
### 彼岸监管国（判刑附属）
### 宗主无特权（无上贡/无征召），监管国无法独立处理任何事物
### 刑期: th_gensokyo_sentence_months（判决脚本设定，20-30 年）
### 刑满: on_monthly 倒计时归零 → 释放为独立国家 + 团结奖励
### ============================================================

th_gensokyo_regulated_state = {
    visible = {
        always = no
    }
    enabled = {
        always = no
    }
    creation_visible = {
        always = no
    }
    subject_creation_enabled = {
        always = no
    }
    release_country_enabled = {
        always = no
    }

    can_attack = {
        always = no
    }
    can_rival = {
        always = no
    }
    can_marry = {
        always = no
    }

    level = 0
    can_be_annexed = no
    has_limited_diplomacy = yes
    subject_can_cancel = no
    overlord_can_cancel = no
    will_join_independence_wars = no
    can_change_rank = no
    can_change_heir_selection = no
    allow_declaring_wars = {
        always = no
    }
    subject_pays = subject_pays_regulated

    overlord_modifier = {
    }
    subject_modifier = {
    }

    on_enable = {
    }

    on_monthly = {
        ### 刑期倒计时（root = subject）
        if = {
            limit = {
                has_variable = th_gensokyo_sentence_months
            }
            change_variable = {
                name = th_gensokyo_sentence_months
                subtract = 1
            }
        }
        ### 刑满释放 + 团结奖励（root.var: 修复 2026-08-07：裸 var: 数值比较报 Event target link）
        if = {
            limit = {
                has_variable = th_gensokyo_sentence_months
                root.var:th_gensokyo_sentence_months <= 0
            }
            ### P1-16 修复：subject_types readme 未给 on_monthly 文档化 scope:overlord；
            ### base game 从附庸 root 取宗主的标准写法是直接链接 overlord（100+ 先例）
            overlord ?= {
                cancel_subject = prev
            }
            international_organization:th_gensokyo_io = {
                th_gensokyo_unity_add_effect = {
                    value = th_unity_release_reward_value
                }
            }
            remove_variable = th_gensokyo_sentence_months
        }
    }
}
