th_gensokyo_trial_effective_sin_effect = {
    scope:crusade_target = {
        set_variable = {
            name = th_gensokyo_effective_sin
            value = var:th_gensokyo_sin_var
        }
        international_organization:th_gensokyo_io = {
            if = {
                limit = {
                    international_organization_has_policy = policy:th_judgment_karma_policy
                }
                scope:crusade_target = {
                    change_variable = {
                        name = th_gensokyo_effective_sin
                        add = 10
                    }
                }
            }
            else_if = {
                limit = {
                    international_organization_has_policy = policy:th_judgment_mercy_policy
                }
                scope:crusade_target = {
                    change_variable = {
                        name = th_gensokyo_effective_sin
                        subtract = 10
                    }
                }
            }
        }
        clamp_variable = {
            name = th_gensokyo_effective_sin
            min = 0
            max = 100
        }
    }
}

            name = th_gensokyo_temp_sentence_years

        change_variable = {
            name = th_gensokyo_temp_sentence_years
            divide = 10

            name = th_gensokyo_temp_sentence_years
            multiply = 12
        }
        change_variable = {
            name = th_gensokyo_sentence_months
            add = var:th_gensokyo_temp_sentence_years
        }
        remove_variable = th_gensokyo_temp_sentence_years
        ### 审判法修正
        international_organization:th_gensokyo_io = {
            if = {
                limit = {
                    international_organization_has_policy = policy:th_judgment_karma_policy

                    change_variable = {
                        name = th_gensokyo_sentence_months

                limit = {
                    international_organization_has_policy = policy:th_judgment_mercy_policy
                }
