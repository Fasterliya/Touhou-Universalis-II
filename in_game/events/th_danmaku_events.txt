### 第 1 轮：对决开始·发起方确认（root = 发起方；双方火力/隐藏终点已由
### th_danmaku_duel_start_effect 就位；2026-08-28 改为确认后进入第 2 轮对方确认）
th_danmaku_events.10 = {
    type = country_event
    title = th_danmaku_events.10.title
    desc = th_danmaku_events.10.desc
    outcome = neutral
    trigger = {
        th_danmaku_duel_in_progress_trigger = yes
    }
    immediate = {
        ### 找回对手 scope（标记法，供 desc 取对手名与触发第 2 轮）
        every_country = {
            limit = {
                has_variable = th_danmaku_duel_opponent_mark
            }
            save_scope_as = th_danmaku_duel_opponent
        }
    }
    ### 礼貌请教：正常倍率 → 第 2 轮对方确认
    option = {
        name = th_danmaku_events.10.a
        ai_chance = {
            base = 60
        }
        custom_tooltip = {
            text = th_danmaku_events.10.a.tt
        }
        set_variable = {
            name = th_danmaku_duel_provoke_var
            value = 0
        }
        if = {
            limit = {
                exists = scope:th_danmaku_duel_opponent
            }
            scope:th_danmaku_duel_opponent = {
                trigger_event_non_silently = th_danmaku_events.11
            }
        }
        else = {
            ### 对手缺失兜底（国家被灭等）：直接取消
            trigger_event_non_silently = th_danmaku_events.12
        }
    }
    ### 挑衅：双倍奖励 / 更严厉惩罚 → 第 2 轮对方确认
    option = {
        name = th_danmaku_events.10.b
        ai_chance = {
            base = 40
        }
        custom_tooltip = {
            text = th_danmaku_events.10.b.tt
        }
        set_variable = {
            name = th_danmaku_duel_provoke_var
            value = 1
        }
        if = {
            limit = {
                exists = scope:th_danmaku_duel_opponent
            }
            scope:th_danmaku_duel_opponent = {
                trigger_event_non_silently = th_danmaku_events.11
            }
        }
        else = {
            trigger_event_non_silently = th_danmaku_events.12
        }
    }
}

### 第 2 轮：对方应战确认（root = 对方；发起方标记/双方火力副本已由 start_effect 就位）
### 接受 → 进入回合循环（.20 ↔ .21）；婉拒 → 发起方收到取消（.12）
th_danmaku_events.11 = {
    type = country_event
    title = th_danmaku_events.11.title
    desc = th_danmaku_events.11.desc
    outcome = neutral
    trigger = {
        has_variable = th_danmaku_duel_opponent_mark
        any_country = {
            has_variable = th_danmaku_duel_challenger_mark
        }
    }
    immediate = {
        ### 找回发起方 scope（供 desc 取名与触发回跳）
        every_country = {
            limit = {
                has_variable = th_danmaku_duel_challenger_mark
            }
            save_scope_as = th_danmaku_duel_challenger
        }
    }
    ### 接受对决 → 第 3 轮起回合循环（root = 发起方）
    option = {
        name = th_danmaku_events.11.a
        ai_chance = {
            base = 95
        }
        custom_tooltip = {
            text = th_danmaku_events.11.a.tt
        }
        if = {
            limit = {
                exists = scope:th_danmaku_duel_challenger
            }
            scope:th_danmaku_duel_challenger = {
                trigger_event_non_silently = th_danmaku_events.20
            }
        }
    }
    ### 婉拒对决 → 发起方收到取消通知
    option = {
        name = th_danmaku_events.11.b
        ai_chance = {
            base = 5
        }
        custom_tooltip = {
            text = th_danmaku_events.11.b.tt
        }
        if = {
            limit = {
                exists = scope:th_danmaku_duel_challenger
            }
            scope:th_danmaku_duel_challenger = {
                trigger_event_non_silently = th_danmaku_events.12
            }
        }
    }
}
