### ============================================================
### 贤者授权扩张 — 裁决事件链（WP-D 2026-08-15 / 重构 2026-08-19）
### .1 贤者裁决（同意/拒绝）  .2 请求者通知（同意）  .3 请求者通知（拒绝）
### 标记法：th_gensokyo_expansion_requester_mark（请求者，挂起）
###         th_gensokyo_expansion_target_owner_mark（目标国）
###         th_gensokyo_expansion_request_mark（目标地块，= CB 依据，保留）
### 结构：option 内 save_scope_as → 调用 scripted_effect（th_gensokyo_expansion_effects.txt）
### 修复 2026-08-19：直接在同 option 引用 save_scope_as 目标报
###   "Undefined event target"（校验期），改为 scripted effect 内引用（审判事件同款已验证模式）。
### 格式：EU5 事件（outcome + option 内直写效果）
### ============================================================

namespace = th_gensokyo_expansion_request_events

### 贤者裁决事件（root = 被请求贤者）
th_gensokyo_expansion_request_events.1 = {
    type = country_event

    title = th_gensokyo_expansion_request_events.1.title
    desc = th_gensokyo_expansion_request_events.1.desc
    outcome = neutral

    trigger = {
        ### 裸脚本触发器需 = yes（EU5 1.3 解析要求）
        th_gensokyo_is_sage_trigger = yes
        any_country = {
            has_variable = th_gensokyo_expansion_requester_mark
        }
    }

    ### 同意：授予请求者针对目标国的 CB（至多 4 地块）
    option = {
        name = th_gensokyo_expansion_request_events.1.a
        ai_chance = {
            base = 90
        }
        every_country = {
            limit = {
                has_variable = th_gensokyo_expansion_requester_mark
            }
            save_scope_as = expansion_requester
        }
        every_country = {
            limit = {
                has_variable = th_gensokyo_expansion_target_owner_mark
            }
            save_scope_as = expansion_target
        }
        th_gensokyo_expansion_grant_cb_effect = yes
        ### 清理挂起/目标国标记（地块标记保留 = CB 依据）
        every_country = {
            limit = {
                OR = {
                    has_variable = th_gensokyo_expansion_requester_mark
                    has_variable = th_gensokyo_expansion_target_owner_mark
                }
            }
            remove_variable = th_gensokyo_expansion_requester_mark
            remove_variable = th_gensokyo_expansion_target_owner_mark
        }
    }

    ### 拒绝：不授予 CB，请求者收敛乱倾向（-5）作为补偿
    option = {
        name = th_gensokyo_expansion_request_events.1.b
        ai_chance = {
            base = 10
        }
        every_country = {
            limit = {
                has_variable = th_gensokyo_expansion_requester_mark
            }
            save_scope_as = expansion_requester
        }
        th_gensokyo_expansion_decline_effect = yes
        ### 清理全部标记（不授予 CB）
        every_country = {
            limit = {
                OR = {
                    has_variable = th_gensokyo_expansion_requester_mark
                    has_variable = th_gensokyo_expansion_target_owner_mark
                }
            }
            remove_variable = th_gensokyo_expansion_requester_mark
            remove_variable = th_gensokyo_expansion_target_owner_mark
        }
    }
}
