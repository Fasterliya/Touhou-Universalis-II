	outcome = neutral

### ------------------------------------------------------------
### 事件 1：鬼王醉酒闹市（事件样板，其余 7 个事件同构待填充）
### 触发：on_monthly random_list 分支 20 权重 + 冷却变量门控
### 机制全放 hidden_effect，玩家只见 custom_tooltip 预览（acgv 规范）
### ------------------------------------------------------------
th_gensokyo_grand_festival.1 = {
	type = country_event
	category = situation_event

	title = th_gensokyo_grand_festival.1.t
	desc = th_gensokyo_grand_festival.1.d
	outcome = neutral

	trigger = {
		NOT = { has_variable = th_gf_ev1_fired }
	}

	immediate = {
		set_variable = {
			name = th_gf_ev1_fired
			months = 6
		}
	}

	option = { ### 花钱安抚：破财消灾，保住分数
		name = th_gensokyo_grand_festival.1.a
		hidden_effect = {
			add_gold = {
				value = -100
			}
		}
		custom_tooltip = th_gensokyo_grand_festival.1.a.tt
	}

	option = { ### 放任不管：分数 -5、稳定小降
		name = th_gensokyo_grand_festival.1.b
		hidden_effect = {
			change_variable = {
				name = th_gf_score
				add = -5
			}
			add_stability = stability_weak_penalty
		}
		custom_tooltip = th_gensokyo_grand_festival.1.b.tt
	}
}
