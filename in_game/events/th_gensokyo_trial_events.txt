### ============================================================
### 阎魔审判事件（玩家为阎魔时手动裁决 3 档）
### 目标国定位: th_gensokyo_trial_target_mark 标记法
### 档位: 事件选项设定 th_gensokyo_sentence_tier（1-3）→ 核心判决
### 3 级模型: 轻=归还领土 / 中=归还+削弱 / 重=监管国（财富转移+40%贡金+归还+削弱）
### ============================================================

namespace = th_gensokyo_trial_events

th_gensokyo_trial_events.1 = {
    type = country_event

    title = th_gensokyo_trial_events.1.t
    desc = th_gensokyo_trial_events.1.d

    trigger = {
        is_leader_of_international_organization = international_organization:th_gensokyo_io
    }

    ### 轻判（档1）：归还非法领土（取消附庸 + 归还核心领土 + 释放国家）
    option = {
        name = th_gensokyo_trial_events.1.a
        effect = {
            every_country = {
                limit = {
                    has_variable = th_gensokyo_trial_target_mark
                }
                save_scope_as = crusade_target
                set_variable = {
                    name = th_gensokyo_sentence_tier
                    value = 1
                }
            }
            every_country = {
                limit = {
                    is_leader_of_international_organization = international_organization:th_gensokyo_io
                }
                save_scope_as = io_leader
            }
            th_gensokyo_trial_judge_core_effect = yes
            set_variable = th_gensokyo_trial_done
        }
    }

    ### 中判（档2）：归还 + 削弱国力
    option = {
        name = th_gensokyo_trial_events.1.b
        effect = {
            every_country = {
                limit = {
                    has_variable = th_gensokyo_trial_target_mark
                }
                save_scope_as = crusade_target
                set_variable = {
                    name = th_gensokyo_sentence_tier
                    value = 2
                }
            }
            every_country = {
                limit = {
                    is_leader_of_international_organization = international_organization:th_gensokyo_io
                }
                save_scope_as = io_leader
            }
            th_gensokyo_trial_judge_core_effect = yes
            set_variable = th_gensokyo_trial_done
        }
    }

    ### 重判（档3）：监管国 —— 财富全转移 + 40% 贡金 + 归还 + 削弱
    option = {
        name = th_gensokyo_trial_events.1.c
        effect = {
            every_country = {
                limit = {
                    has_variable = th_gensokyo_trial_target_mark
                }
                save_scope_as = crusade_target
                set_variable = {
                    name = th_gensokyo_sentence_tier
                    value = 3
                }
            }
            every_country = {
                limit = {
                    is_leader_of_international_organization = international_organization:th_gensokyo_io
                }
                save_scope_as = io_leader
            }
            th_gensokyo_trial_judge_core_effect = yes
            set_variable = th_gensokyo_trial_done
        }
    }
}
