### ============================================================
### 阎魔审判事件（玩家为阎魔时手动裁决 4 档）
### 目标国定位: th_gensokyo_trial_target_mark 标记法
### 档位: 事件选项设定 th_gensokyo_sentence_tier（0-3）→ 核心判决
### 档0 仅赔款破产 / 档1 归还领土 / 档2 +削弱国力 / 档3 +监管国（财富转移+40%贡金+归还+削弱）
### 格式修复 2026-08-19：EU5 事件需 outcome = ...；option 内效果直写（无 effect 包裹）
### ============================================================

namespace = th_gensokyo_trial_events

th_gensokyo_trial_events.1 = {
    type = country_event

    title = th_gensokyo_trial_events.1.title
    desc = th_gensokyo_trial_events.1.desc
    outcome = neutral

    trigger = {
        is_leader_of_international_organization = international_organization:th_gensokyo_io
    }

    ### 轻判（档1）：归还非法领土（取消附庸 + 归还核心领土 + 释放国家）
    option = {
        name = th_gensokyo_trial_events.1.a
        every_country = {
            limit = {
                has_variable = th_gensokyo_trial_target_mark
            }
            save_scope_as = crusade_target
            set_variable = {
                name = th_gensokyo_sentence_tier
                value = 1
            }
        }
        every_country = {
            limit = {
                is_leader_of_international_organization = international_organization:th_gensokyo_io
            }
            save_scope_as = io_leader
        }
        th_gensokyo_trial_judge_core_effect = yes
        set_variable = th_gensokyo_trial_done
    }

    ### 中判（档2）：归还 + 削弱国力
    option = {
        name = th_gensokyo_trial_events.1.b
        every_country = {
            limit = {
                has_variable = th_gensokyo_trial_target_mark
            }
            save_scope_as = crusade_target
            set_variable = {
                name = th_gensokyo_sentence_tier
                value = 2
            }
        }
        every_country = {
            limit = {
                is_leader_of_international_organization = international_organization:th_gensokyo_io
            }
            save_scope_as = io_leader
        }
        th_gensokyo_trial_judge_core_effect = yes
        set_variable = th_gensokyo_trial_done
    }

    ### 重判（档3）：监管国 —— 财富全转移 + 40% 贡金 + 归还 + 削弱
    option = {
        name = th_gensokyo_trial_events.1.c
        every_country = {
            limit = {
                has_variable = th_gensokyo_trial_target_mark
            }
            save_scope_as = crusade_target
            set_variable = {
                name = th_gensokyo_sentence_tier
                value = 3
            }
        }
        every_country = {
            limit = {
                is_leader_of_international_organization = international_organization:th_gensokyo_io
            }
            save_scope_as = io_leader
        }
        th_gensokyo_trial_judge_core_effect = yes
        set_variable = th_gensokyo_trial_done
    }

    ### WP-H 2026-08-15（档0）：仅赔款破产 —— 转移半数国库 + 破产 debuff（2 年），不归还领土
    ### 注：选项键从 .e 起（.d 已被旧 desc 惯例占用；现 desc 为 .desc）
    option = {
        name = th_gensokyo_trial_events.1.e
        every_country = {
            limit = {
                has_variable = th_gensokyo_trial_target_mark
            }
            save_scope_as = crusade_target
            set_variable = {
                name = th_gensokyo_sentence_tier
                value = 0
            }
        }
        every_country = {
            limit = {
                is_leader_of_international_organization = international_organization:th_gensokyo_io
            }
            save_scope_as = io_leader
        }
        th_gensokyo_trial_judge_core_effect = yes
        set_variable = th_gensokyo_trial_done
    }
}
