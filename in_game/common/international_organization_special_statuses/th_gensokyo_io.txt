### 幻想乡国际组织 — 特殊席位定义
### 字段顺序: max_countries → priority → can_bestow_trigger → auto_bestowal_trigger → auto_rescind_trigger
###           → modifier → leader_modifier → on_bestowed_effect → on_rescinded_effect
###           → leader/elector → special_status_power → map_color
### 参考: base game hre.txt lordship_of_ireland.txt high_kingship.txt japanese_shogunate.txt

### ============================================================
### 席位5：幻想乡席位（占位验证）th_gensokyo_io_seat_special_status
### 对标 base game union.txt senior_partner — 政策门控 + 自动授衔
### 政策 th_gensokyo_seat_unlock_policy 激活 → 解锁自动授予
### 政策移除 → 自动收回（auto_rescind，无 OR）
### 授予条件：成员 + 政策激活 + 名额空 + 个人贡献度最高者
### ============================================================
th_gensokyo_io_seat_special_status = {
	max_countries = 1
	priority = 5

	can_bestow_trigger = {
		is_member_of_international_organization = scope:recipient
		scope:recipient = {
			international_organization_has_policy = policy:th_gensokyo_seat_unlock_policy
			"num_countries_with_special_status(special_status:th_gensokyo_io_seat_special_status)" < 1
			NOT = {
				any_international_organization_member = {
					this != root
					var:th_gensokyo_personal_contribution_var > root.var:th_gensokyo_personal_contribution_var
				}
			}
		}
	}
	auto_bestowal_trigger = {
		always = yes
	}
	auto_rescind_trigger = {
		NOT = {
			scope:recipient = {
				international_organization_has_policy = policy:th_gensokyo_seat_unlock_policy
			}
		}
	}

	modifier = {
		monthly_prestige = 0.1
		# TODO: 占位修正值
	}
	### 2026-08-21：投票权重统一改幻想乡影响力动态公式（原 0 = 无票权）
	special_status_power = {
		value = th_gensokyo_io_vote_power_value
	}
	map_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TIER3_COLOR
}

### ============================================================
### 席位6：幻想乡普通成员 th_gensokyo_io_regular_member_special_status
### 2026-08-21 新增（bug 1 修复：普通国家无投票权）
### 根因：95 个成员中仅 26 个有特殊席位（贤者 3/人类村落 20/阎魔 1/守护 1/占位 1），
###       无席位国家 country_highest_rated_special_status_power = 0 → 议会权重恒 0
### 本席位 = 全体成员基础席位（auto_bestowal 全员授予），提供动态投票权重
### 权重公式: th_gensokyo_io_vote_power_value（影响力 × (1+缘乱修正)）
### 与专属席位并存：country_highest_rated_special_status_power 取最高值，
### 各席位公式相同 → 无冲突（参考 base game union.txt junior_partner 动态 power 模式）
### ============================================================
th_gensokyo_io_regular_member_special_status = {
	max_countries = 95
	priority = 4

	can_bestow_trigger = {
		is_member_of_international_organization = scope:recipient
	}
	### 全员自动授予（base game auto_bestowal 先例：hre/union/high_kingship 等 10 文件）
	auto_bestowal_trigger = {
		always = yes
	}
	### 离开 IO 时收回（IO can_leave = no，实际不触发）
	auto_rescind_trigger = {
		NOT = {
			is_member_of_international_organization = scope:recipient
		}
	}

	modifier = {
	}
	special_status_power = {
		value = th_gensokyo_io_vote_power_value
	}
	map_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TIER3_COLOR
}
