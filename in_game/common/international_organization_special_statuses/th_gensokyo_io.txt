### 幻想乡国际组织 — 特殊席位定义
### 字段顺序: max_countries → priority → can_bestow_trigger → auto_bestowal_trigger → auto_rescind_trigger
###           → modifier → leader_modifier → on_bestowed_effect → on_rescinded_effect
###           → leader/elector → special_status_power → map_color
### 参考: base game hre.txt lordship_of_ireland.txt high_kingship.txt japanese_shogunate.txt

### ============================================================
### 席位1：阎魔（IO领袖）th_gensokyo_io_yama_special_status
### 对标 HRE emperor (hre.txt:L1-39) + high_king (high_kingship.txt:L1-21)
### ============================================================
th_gensokyo_io_yama_special_status = {
    max_countries = 1
    priority = 15

    can_bestow_trigger = {
        is_leader_of_international_organization = scope:recipient
    }
    auto_bestowal_trigger = {
        always = yes
    }
    auto_rescind_trigger = {
        NOT = { is_leader_of_international_organization = scope:recipient }
    }

    modifier = {
        allowed_intervene_in_war = yes
        allowed_enforce_peace = yes
        monthly_prestige = 0.10
        # TODO: 后续人工填写阎魔席位修正值
    }

    on_bestowed_effect = {
    }
    on_rescinded_effect = {
    }

    leader = yes
    special_status_power = {
        value = 200
    }
    map_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_LEADER_COLOR
}

### 席位2：幻想乡守护 th_gensokyo_io_guardian_special_status
### 对标 HRE elector (hre.txt:L40-115) — 仅手动+开局预分配，永不自动
### ============================================================
th_gensokyo_io_guardian_special_status = {
    max_countries = 1
    priority = 10

    can_bestow_trigger = {
        is_member_of_international_organization = scope:recipient
        scope:recipient = {
            NOT = {
                country_has_special_status = { type = special_status:th_gensokyo_io_guardian_special_status country = root }
            }
        }
    }
    auto_bestowal_trigger = {
    }
    auto_rescind_trigger = {
        NOT = { is_member_of_international_organization = scope:recipient }
    }

    modifier = {
        global_defensiveness_modifier = 0.10
        monthly_prestige = 0.05
        # TODO: 后续人工填写守护席位修正值
    }

    on_bestowed_effect = {
    }
    on_rescinded_effect = {
    }

    special_status_power = {
        value = 180
    }
    map_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TIER1_COLOR
