th_gensokyo_unlawful_expansion_check = {
    trigger = {
        scope:winner = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        scope:loser = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }
    effect = {
        if = {
            limit = {
                th_gensokyo_has_authorized_expansion_trigger
            }
            ### 合法扩张：消耗配额 1 块 + 少量趋向好战（B6：敌意镜像，量小）
            change_variable = {
                name = th_gensokyo_expansion_quota_used
                add = 1
            }
            th_gensokyo_values_shift_belligerent_effect = {
                value = th_expansion_legal_notorious_value
            }
        }
        else = {
            ### 非法扩张：惩罚链（恶名量更大，B6）
            scope:loser = {
                save_scope_as = expansion_victim
            }
            th_gensokyo_expansion_punish_effect = {
                value = th_expansion_unlawful_notorious_value
                unity = th_expansion_unlawful_unity_value
            }
            change_variable = {
                name = th_gensokyo_expansion_quota_used
                add = 1
            }
        }
    }
}

th_gensokyo_sin_battle_check = {
    trigger = {
        has_variable = th_gensokyo_crusade_target_mark
    }
    effect = {
        change_variable = {
            name = th_gensokyo_sin_var
            add = scope:war_score
        }
        clamp_variable = {
            name = th_gensokyo_sin_var
            min = 0
            max = 100
        }
    }
}


th_gensokyo_sin_refusal_check = {
    trigger = {
        OR = {
            has_variable = th_gensokyo_crusade_target_mark
            scope:recipient = {
                has_variable = th_gensokyo_crusade_target_mark
            }
        }
    }
    effect = {
        if = {
            limit = {
                has_variable = th_gensokyo_crusade_target_mark
            }
            change_variable = {
                name = th_gensokyo_sin_var
                add = th_crusade_sin_refusal_value
            }
            clamp_variable = {
                name = th_gensokyo_sin_var
                min = 0
                max = 100
            }
        }
        else_if = {
            limit = {
                scope:recipient = {
                    has_variable = th_gensokyo_crusade_target_mark
                }
            }
            scope:recipient = {
                change_variable = {
                    name = th_gensokyo_sin_var
                    add = th_crusade_sin_refusal_value
                }
                clamp_variable = {
                    name = th_gensokyo_sin_var
                    min = 0
                    max = 100
                }
            }
        }
    }
}


### === 审判庭宣战 → 激活惩戒局势（2026-08-23 重构） ===
### 用户决定：局势（战争界面）在审判庭对目标国宣战之后弹出——
### 任何审判庭成员用惩戒 CB 对目标宣战都会激活局势（玩家发起与 AI 后续宣战同路径）
### 2026-08-26：target 机制弃用 → 目标判定改 th_gensokyo_crusade_target_mark 标记
### （原 international_organization:th_gensokyo_tribunal = { target ?= scope:recipient }
###  在 target 恒空时钩子恒 false → 局势永不激活）
th_gensokyo_tribunal_war_activate = {
    trigger = {
        ### 宣战国是审判庭成员
        scope:actor = {
            is_member_of_international_organization = international_organization:th_gensokyo_tribunal
        }
        ### 被宣战国是惩戒目标（标记法）
        scope:recipient = {
            has_variable = th_gensokyo_crusade_target_mark
        }
        ### 惩戒 CB 战争（其它战争不激活）
        scope:war = {
            casus_belli ?= casus_belli:th_gensokyo_crusade_cb
        }
        ### 局势未激活时再激活（防重复调用刷屏）
        NOT = {
            situation:th_gensokyo_crusade = {
                situation_is_active = yes
            }
        }
    }
    effect = {
        ### 2026-08-26（用户补充）：宣战后召唤审判庭全体成员加入对目标战争
        ### （join_offensive_wars_always 引擎层双保险；scope:war 实证可用）
        international_organization:th_gensokyo_tribunal = {
            every_international_organization_member = {
                limit = {
                    NOT = {
                        this = scope:actor
                    }
                    NOT = {
                        is_at_war_with = scope:recipient
                    }
                }
                join_war_as_attacker = {
                    war = scope:war
                    call_in_subjects = yes
                }
            }
        }
        international_organization:th_gensokyo_tribunal = {
            leader_country ?= {
                enable_situation_effect = {
                    type = th_gensokyo_crusade
                }
                activate_situation = situation:th_gensokyo_crusade
            }
        }
    }
}
