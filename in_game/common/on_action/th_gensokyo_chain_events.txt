th_gensokyo_unlawful_expansion_check = {
    trigger = {
        scope:winner = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        scope:loser = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }
    effect = {
        if = {
            limit = {
                th_gensokyo_has_authorized_expansion_trigger
            }
            ### 合法扩张：消耗配额 1 块 + 少量趋向好战（B6：敌意镜像，量小）
            change_variable = {
                name = th_gensokyo_expansion_quota_used
                add = 1
            }
            th_gensokyo_values_shift_belligerent_effect = {
                value = th_expansion_legal_notorious_value
            }
            ### 2026-10-01 T2-1 罪名账本：合法扩张 +1 次
            ### 罪名为「合法扩张」——**不计入罪值合计**，仅供 T2-2 立案后自辩举证
            ### （铁律 11：记账走 has_variable 守卫的共享 effect）
            th_gensokyo_crime_record_effect = {
                crime = th_gensokyo_crime_legal_var
            }
        }
        else = {
            ### 非法扩张：惩罚链（恶名量更大，B6）
            scope:loser = {
                save_scope_as = expansion_victim
            }
            th_gensokyo_expansion_punish_effect = {
                value = th_expansion_unlawful_notorious_value
                unity = th_expansion_unlawful_unity_value
            }
            ### ============================================================
            ### 2026-10-01 T2-2-④：涉案地块 + 受害国入物证（起诉书用）
            ### 依据 = 本钩子的官方 scope 契约，逐字见
            ###   VANILLA\common\on_action\_hardcoded.txt:2341
            ###   "#root = winner, scope:location = location, scope:winner = new owner,
            ###    scope:loser = old owner"
            ###   （同文件 :2347-2369 即该契约的逐字用法先例：scope:location 直接
            ###    作为 add_to_global_variable_list 的 target）
            ### 为什么记在这里而不是立案时：T2-1 的记账点只累加**次数**，
            ###   "地块"只在本钩子一次性可得（《脚本规格》§6.1-4 的结论）。
            ###   立案时若 clear_global_variable_list 重填，等于销毁起诉书 →
            ###   本窗立案行动**不清理**该列表（见 th_gensokyo_tribunal_indict.txt ④ 注释）。
            ### 写入门：只在作案者**尚无在办案件**（无 tarnish）时记录 ⇒
            ###   立案之后该国的继续作案不会污染"本案物证"；
            ###   旧案物证的清理由结案路径（T2-2b / T2-3）与开局 init 负责。
            ### 去重：同一地块可能因多个钩子重复触发，用
            ###   is_target_in_global_variable_list 先查（触发器存在性：
            ###   VANILLA\common\trigger_localization\common_triggers.txt:130；
            ###   用法先例 events\disaster\hook_and_cod_wars.txt:1300）
            ### 本 on_action 的 trigger 已保证 scope:winner / scope:loser 都是
            ###   幻想乡 IO 成员 ⇒ 物证天然限定在幻想乡相关国家
            ### ============================================================
            if = {
                limit = {
                    exists = scope:location
                    NOT = { has_variable = th_gensokyo_tarnish_mark }
                    NOT = {
                        is_target_in_global_variable_list = {
                            name = th_gensokyo_stolen_location_list
                            target = scope:location
                        }
                    }
                }
                add_to_global_variable_list = {
                    name = th_gensokyo_stolen_location_list
                    target = scope:location
                }
                ### 受害国标记（?= 空链接守卫；被吞并的旧主可能已不存在）
                scope:loser ?= {
                    set_variable = th_gensokyo_victim_mark
                }
            }
            change_variable = {
                name = th_gensokyo_expansion_quota_used
                add = 1
            }
        }
    }
}

th_gensokyo_sin_battle_check = {
    trigger = {
        has_variable = th_gensokyo_crusade_target_mark
    }
    effect = {
        ### 2026-10-01 T2-3 罪 / 态度分流：战时累积改指向 th_gensokyo_attitude_var
        ### （原写 th_gensokyo_sin_var）。th_gensokyo_sin_var 语义收窄为「罪」，
        ### 今后只由立案时按罪名账本写入、上限 th_sin_crime_cap；本窗不改判决读法（属 T2-4）。

            limit = {
                has_variable = th_gensokyo_attitude_var

                name = th_gensokyo_attitude_var
                add = scope:war_score
            }
            clamp_variable = {
                name = th_gensokyo_attitude_var
                min = 0
                max = 40
            }
        }
    }
}


th_gensokyo_sin_refusal_check = {
    trigger = {
        OR = {
            has_variable = th_gensokyo_crusade_target_mark

                has_variable = th_gensokyo_crusade_target_mark
            }
        }
    }
    effect = {
        ### 2026-10-01 T2-3 罪 / 态度分流：两个分支的战时累积都改指向
        ### th_gensokyo_attitude_var（原写 th_gensokyo_sin_var），上限 th_attitude_cap = 40。
        ### th_gensokyo_sin_var 语义收窄为「罪」，今后只由立案时按账本写入（T2-2）；
        ### 本窗不改判决读法（属 T2-4）。两处都加 has_variable 守卫（铁律 11）。
        if = {
            limit = {
                has_variable = th_gensokyo_crusade_target_mark
                has_variable = th_gensokyo_attitude_var
            }
            change_variable = {
                name = th_gensokyo_attitude_var
                add = th_crusade_sin_refusal_value
            }
            clamp_variable = {
                name = th_gensokyo_attitude_var
                min = 0
                max = 40

        else_if = {
            limit = {
                scope:recipient = {
                    has_variable = th_gensokyo_crusade_target_mark
                    has_variable = th_gensokyo_attitude_var
                }
            }
            scope:recipient = {
                change_variable = {
                    name = th_gensokyo_attitude_var
                    add = th_crusade_sin_refusal_value
                }
                clamp_variable = {
                    name = th_gensokyo_attitude_var

    }

        }

            has_variable = th_gensokyo_crusade_target_mark
        }
    }
    effect = {
        ### 召唤审判庭全体成员加入对目标战争（join_offensive_wars_always

                    NOT = {
                        this = scope:actor

                    NOT = {
                        is_at_war_with = scope:recipient
                    }
                }
                join_war_as_attacker = {
                    war = scope:war
                    call_in_subjects = yes
                }
            }
        }
    }
}
