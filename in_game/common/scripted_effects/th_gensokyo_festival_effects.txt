### 参与者初始化：本国所有市场挂祭典临时需求（5 倍基础酒需求，调参点）
th_gf_setup_participant_effect = {
	every_market_present_in_country = {
		limit = {
			NOT = { has_temporary_demand = demand:th_gensokyo_grand_festival_wine }
		}
		add_temporary_demand = {
			type = demand:th_gensokyo_grand_festival_wine
			scale_with_pop = 0.002
			years = -1
		}
	}
}

### 分档奖励（声望/稳定/亲善价值观）
### 数值出处：default_values.txt:26-35（prestige_*）与 :10-21（stability_*）
### 价值观移动复用模组 th_gensokyo_feast.txt:100 同构写法
th_gf_reward_effect = {
	if = {
		limit = { var:th_gf_score >= 70 }
		add_prestige = prestige_severe_bonus
		add_stability = stability_mild_bonus
		change_societal_value = {
			type = th_gensokyo_amity_vs_belligerence
			value = th_feast_notorious_reduction_value
		}
	}
	else_if = {
		limit = { var:th_gf_score >= 40 }
		add_prestige = prestige_mild_bonus
		add_stability = stability_mild_bonus
	}
	else_if = {
		limit = { var:th_gf_score >= 20 }
		add_stability = stability_weak_bonus
	}
}
