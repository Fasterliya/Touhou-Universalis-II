### 幻想乡团结度 增加（在IO scope上调用，设计§2.1）
th_gensokyo_io_unity_var_add_effect = {
    change_variable = {
        name = th_gensokyo_unity_var
        add = $value$
    }
    clamp_variable = {
        name = th_gensokyo_unity_var
        min = 0
        max = 100
    }
}

### 幻想乡个人贡献度 增加（在国家scope上调用，设计§2.3）
### 附带恶名惩罚：恶名≥30 → -30%，≥50 → -50%，≥70 → -70%（设计§2.5）
th_gensokyo_io_contribution_var_add_effect = {
    set_variable = {
        name = th_gensokyo_temp_contribution_gain
        value = $value$
    }
    if = {
        limit = {
            var:th_gensokyo_notorious_reputation_var >= 70
        }
        change_variable = {
            name = th_gensokyo_temp_contribution_gain
            multiply = 0.3
        }
    }
    else_if = {
        limit = {
            var:th_gensokyo_notorious_reputation_var >= 50
        }
        change_variable = {
            name = th_gensokyo_temp_contribution_gain
            multiply = 0.5
        }
    }
    else_if = {
        limit = {
            var:th_gensokyo_notorious_reputation_var >= 30
        }
        change_variable = {
            name = th_gensokyo_temp_contribution_gain
            multiply = 0.7
        }
    }
    change_variable = {
        name = th_gensokyo_personal_contribution_var
        add = var:th_gensokyo_temp_contribution_gain
    }
}
