th_gensokyo_crusade_debate_start_effect = {
    ### 定位目标国（标记法）
    every_international_organization_member = {
        limit = {
            has_variable = th_gensokyo_crusade_target_mark
        }
        save_scope_as = crusade_target
    }
    if = {
        limit = {
            exists = scope:crusade_target
        }
        ### ① 全员预投反对（v3：所有人都会投票，无人"尚未投票"）
        ### set_participated_in_parliament 不改变支持度（base game propose 同款）
        every_international_organization_member = {
            set_participated_in_parliament = {
                international_organization = international_organization:th_gensokyo_io
                vote = no
            }
        }
        ### === 赞成组 ===
        ### ② 目标国敌人/宿敌（succession_crisis_actions.txt:143 触发器实证）
        every_international_organization_member = {
            limit = {
                NOT = {
                    has_variable = th_gensokyo_crusade_vote_locked
                }
                OR = {
                    is_rival_of = scope:crusade_target
                    is_enemy_of = scope:crusade_target
                }
            }
            participate_in_parliament = {
                international_organization = international_organization:th_gensokyo_io
                vote = yes
            }
            set_variable = th_gensokyo_crusade_vote_locked
            ### 同意者标记（加入倾向）
            set_variable = th_gensokyo_crusade_supporter
        }
        ### ③ 针对目标国的原版 coalition（包围网）成员（q1 决策：
        ###    「判定是否有针对开会目标的包围网，把他们全部加入赞同」）

        every_international_organization = {
            limit = {
                international_organization_type = international_organization_type:coalition
                international_organization_target ?= scope:crusade_target
            }
            every_international_organization_member = {
                limit = {
                    NOT = {
                        has_variable = th_gensokyo_crusade_vote_locked
                    }
                    is_member_of_international_organization = international_organization:th_gensokyo_io
                }
                participate_in_parliament = {
                    international_organization = international_organization:th_gensokyo_io
                    vote = yes
                }
                set_variable = th_gensokyo_crusade_vote_locked
                ### 同意者标记（加入倾向）
                set_variable = th_gensokyo_crusade_supporter
            }
        }
        ### ④ 贤者
        every_international_organization_member = {
            limit = {
                NOT = {
                    has_variable = th_gensokyo_crusade_vote_locked
                }
                th_gensokyo_is_sage_trigger = yes
            }
            participate_in_parliament = {
                international_organization = international_organization:th_gensokyo_io
                vote = yes
            }
            set_variable = th_gensokyo_crusade_vote_locked
            ### 同意者标记（加入倾向）
            set_variable = th_gensokyo_crusade_supporter
        }
        ### ⑤ 领袖（发起人阎魔）——发起人必赞成（v3 追加）
        every_international_organization_member = {
            limit = {
                NOT = {

            participate_in_parliament = {
                international_organization = international_organization:th_gensokyo_io
