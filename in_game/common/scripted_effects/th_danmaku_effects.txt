### 遭遇战四结局结算（需 th_danmaku_firepower_self_var /
### th_danmaku_encounter_youkai_fp_var 就位；国家 scope）
### 算法：火力比（我方×100÷妖怪，整数%）分五档，每档给四结局权重 random_list
th_danmaku_encounter_resolve_effect = {
    set_variable = {
        name = th_danmaku_encounter_ratio_pct_var
        value = th_danmaku_encounter_ratio_pct_value
    }
    ### 除零/极低妖怪火力兜底（妖怪火力必有下限，此处防御）
    if = {
        limit = {
            var:th_danmaku_encounter_youkai_fp_var <= 0
        }
        set_variable = {
            name = th_danmaku_encounter_ratio_pct_var
            value = 200
        }
    }
    if = {
        limit = { var:th_danmaku_encounter_ratio_pct_var >= 160 }
        th_danmaku_encounter_outcome_roll_effect = { great_v = 45 victory = 40 defeat = 12 great_defeat = 3 }
    }
    else_if = {
        limit = { var:th_danmaku_encounter_ratio_pct_var >= 120 }
        th_danmaku_encounter_outcome_roll_effect = { great_v = 35 victory = 40 defeat = 20 great_defeat = 5 }
    }
    else_if = {
        limit = { var:th_danmaku_encounter_ratio_pct_var >= 90 }
        th_danmaku_encounter_outcome_roll_effect = { great_v = 20 victory = 35 defeat = 30 great_defeat = 15 }
    }
    else_if = {
        limit = { var:th_danmaku_encounter_ratio_pct_var >= 60 }
        th_danmaku_encounter_outcome_roll_effect = { great_v = 10 victory = 25 defeat = 35 great_defeat = 30 }
    }
    else = {
        th_danmaku_encounter_outcome_roll_effect = { great_v = 5 victory = 15 defeat = 35 great_defeat = 45 }
    }
}

### 对决战随机选人并开始（显式发起方参数；行动/事件随时可调）
### 用法: th_danmaku_duel_trigger_effect = { challenger = scope:actor }（行动）
###       th_danmaku_duel_trigger_effect = { challenger = ROOT }（事件）
### 选人优先级：异变参与者列表 → 退治者标记 → 幻想乡 IO 成员（兜底）
### 列表/标记为 M3 异变系统预留，未填充时自动回落，不会报错
### 2026-08-28：不依赖隐式作用域（行动 effect 顶层 root 为空）——排除条件用
###   this = $challenger$ 显式比较，透传 challenger 给 th_danmaku_duel_start_effect
th_danmaku_duel_trigger_effect = {
    ### 1) 异变参与者（global_variable_list:th_gensokyo_incident_participant_list）
    if = {
        limit = {
            exists = $challenger$
            th_gensokyo_in_incident_trigger = yes
        }
        random_country = {
            limit = {
                NOT = { this = $challenger$ }
                is_target_in_global_variable_list = {
                    name = th_gensokyo_incident_participant_list
                    target = this
                }
                NOT = { th_danmaku_duel_in_progress_trigger = yes }
            }
            save_scope_as = th_danmaku_duel_opponent_scope
        }
    }
    ### 2) 退治者（预留标记 th_gensokyo_incident_resolver_mark）
    if = {
        limit = {
            exists = $challenger$
            th_gensokyo_in_incident_trigger = yes
            NOT = { exists = scope:th_danmaku_duel_opponent_scope }
        }
        random_country = {
            limit = {
                NOT = { this = $challenger$ }
                has_variable = th_gensokyo_incident_resolver_mark
                NOT = { th_danmaku_duel_in_progress_trigger = yes }
            }
            save_scope_as = th_danmaku_duel_opponent_scope
        }
    }
    ### 3) 兜底：幻想乡 IO 任意成员
    if = {
        limit = {
            exists = $challenger$
            NOT = { exists = scope:th_danmaku_duel_opponent_scope }
        }
        random_country = {
            limit = {
                NOT = { this = $challenger$ }
                is_member_of_international_organization = international_organization:th_gensokyo_io
                NOT = { th_danmaku_duel_in_progress_trigger = yes }
            }
            save_scope_as = th_danmaku_duel_opponent_scope
        }
    }
    ### 选到 → 开始对决（显式透传发起方）
    if = {
        limit = {
            exists = $challenger$
            exists = scope:th_danmaku_duel_opponent_scope
        }
        th_danmaku_duel_start_effect = {
            challenger = $challenger$
            target = scope:th_danmaku_duel_opponent_scope
        }
        clear_saved_scope = th_danmaku_duel_opponent_scope
    }
}

### 对决战开始（显式发起方/对手参数；行动/事件/接口均可调）
### 用法: th_danmaku_duel_start_effect = { challenger = scope:actor target = scope:target }
###       th_danmaku_duel_start_effect = { challenger = ROOT target = scope:某国 }
### 守卫：双方存在 + 均非对决中；不满足则安全退出
### 2026-08-28 作用域修复：不依赖任何隐式 ROOT（行动 effect 顶层 root 为空，P1-12 教训）；
###   发起方/对手操作全部放进显式 $challenger$/$target$ 块，跨国家读变量用
###   $target$.var:X / $challenger$.var:X（替换后即 scope:target.var:X，参考库已验证），
###   不使用 prev / 保存作用域（模组 F24 教训：跨块保存作用域不可靠）。
### 流程：发起方持全部状态（self FP / 对手 FP 拷回 / goal / 分数 / 回合 / 挑衅）
###   → 对手持 mark + 双方火力副本 → 触发 .10 第 1 轮（发起方确认）
th_danmaku_duel_start_effect = {
    if = {
        limit = {
            exists = $challenger$
            exists = $target$
            $challenger$ = {
                NOT = { th_danmaku_duel_in_progress_trigger = yes }
            }
            $target$ = {
                NOT = { th_danmaku_duel_in_progress_trigger = yes }
            }
        }
        ### 发起方：火力快照 + 发起方火力副本 + 发起方标记（供对方 .11 预览与回找）
        $challenger$ = {
            th_danmaku_firepower_snapshot_effect = yes
            set_variable = {
                name = th_danmaku_duel_challenger_fp_var
                value = var:th_danmaku_firepower_self_var
            }
            set_variable = {
                name = th_danmaku_duel_challenger_mark
                value = 1
            }
        }
        ### 对手：自身火力 + 发起方火力副本（显式读 $challenger$.var）+ 对手标记
        $target$ = {
            set_variable = {
                name = th_danmaku_duel_enemy_fp_var
                value = th_danmaku_firepower_value
            }
            set_variable = {
                name = th_danmaku_duel_challenger_fp_var
                value = $challenger$.var:th_danmaku_duel_challenger_fp_var
            }
            set_variable = {
                name = th_danmaku_duel_opponent_mark
                value = 1
            }
        }
        ### 发起方：对手火力拷回（goal 公式与 .10/.20 loc 在发起方读取）+ 隐藏终点 + 状态
        $challenger$ = {
            set_variable = {
                name = th_danmaku_duel_enemy_fp_var
                value = $target$.var:th_danmaku_duel_enemy_fp_var
            }
            set_variable = {
                name = th_danmaku_duel_goal_var
                value = th_danmaku_duel_goal_value
            }
            set_variable = { name = th_danmaku_duel_player_score_var value = 0 }
            set_variable = { name = th_danmaku_duel_enemy_score_var value = 0 }
            set_variable = { name = th_danmaku_duel_round_var value = 1 }
            set_variable = { name = th_danmaku_duel_provoke_var value = 0 }
            set_variable = { name = th_danmaku_duel_last_player_gain_var value = 0 }
            set_variable = { name = th_danmaku_duel_last_enemy_gain_var value = 0 }
            set_variable = {
                name = th_danmaku_duel_deadline_months_var
                value = th_danmaku_duel_stale_months_value
            }
            ### 第 1 轮：发起方确认事件（礼貌请教 / 挑衅），TT 预览双方火力
            trigger_event_non_silently = th_danmaku_events.10
        }
    }
}
