### 决斗总结算（应战方 root 作用域调用）
th_danmaku_resolve_duel_effect = {
    ### 挑战者火力 → 存入 root 的临时变量
    var:th_danmaku_challenger_var = {
        root = {
            set_variable = {
                name = th_dfp_enemy
                value = th_danmaku_firepower_value
            }
        }
    }
    ### 应战方（root）火力
    set_variable = {
        name = th_dfp_self
        value = th_danmaku_firepower_value
    }

    ### 应战方胜
    if = {
        limit = {
            var:th_dfp_self > var:th_dfp_enemy
        }
        th_danmaku_duel_winner_effect = yes
        var:th_danmaku_challenger_var = {
            th_danmaku_duel_loser_effect = yes
        }
        trigger_event_non_silently = th_danmaku.2
        var:th_danmaku_challenger_var = {
            trigger_event_non_silently = th_danmaku.3
        }
    }
    ### 挑战者胜
    else_if = {
        limit = {
            var:th_dfp_enemy > var:th_dfp_self
        }
        th_danmaku_duel_loser_effect = yes
        var:th_danmaku_challenger_var = {
            th_danmaku_duel_winner_effect = yes
        }
        trigger_event_non_silently = th_danmaku.3
        var:th_danmaku_challenger_var = {
            trigger_event_non_silently = th_danmaku.2
        }
    }
    ### 平局（火力相等：双方小奖励）
    else = {
        th_danmaku_duel_draw_effect = yes
        var:th_danmaku_challenger_var = {
            th_danmaku_duel_draw_effect = yes
        }
        trigger_event_non_silently = th_danmaku.4
        var:th_danmaku_challenger_var = {
            trigger_event_non_silently = th_danmaku.4
        }
    }

    ### 清理（先清挑战者侧，再清自身，顺序不可颠倒）
    var:th_danmaku_challenger_var = {
        remove_variable = th_danmaku_pending_challenge_target_var
    }
    remove_variable = th_danmaku_challenger_var
    remove_variable = th_dfp_self
    remove_variable = th_dfp_enemy
}
