### 罪名记账（T2-1 新增，2026-10-01）
### root = 记账对象国（扩张者 / 吞并者 / 宣战方）；$crime$ = 写入的罪名变量名
### 每次 +1，同时覆盖"最近一次记账年份"
### 用法: th_gensokyo_crime_record_effect = { crime = th_gensokyo_crime_unlawful_var }
### $param$ 当变量名用的蓝本：vanilla scripted_effects/global_effects.txt:89-94
###   （limit_variable_to_max = { if = { limit = { var:$name$ > $value$ }
###      set_variable = { name = $name$ value = $value$ } } }）
### 时间戳记"年份"而非"年月数"：current_year 有存进变量的 vanilla 先例
###   （government_reforms/country_specific.txt:501-505），current_month 全库只作比较、
###   无作值先例 → 年月数写法 FAILED，退化为年份（见 T2 报告 §Verification 6）
### ============================================================
th_gensokyo_crime_record_effect = {
    if = {
        limit = {
            has_variable = $crime$
        }
        change_variable = {
            name = $crime$
            add = 1
        }
    }
    else = {
        set_variable = {
            name = $crime$
            value = 1
        }
    }
    set_variable = {
        name = th_gensokyo_crime_last_date
        value = current_year
    }
}
