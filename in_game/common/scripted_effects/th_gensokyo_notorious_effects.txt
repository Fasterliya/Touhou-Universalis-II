th_gensokyo_chain_init_effect = {
    every_country = {
        limit = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        ### 2026-08-14 首月空窗修复：原恶名/贡献/影响力只靠月度兜底补设，
        ### 开局首月点宴会等行动报 S11 家族 "Variable not of the 'value' scope type.

        }

        set_variable = {
