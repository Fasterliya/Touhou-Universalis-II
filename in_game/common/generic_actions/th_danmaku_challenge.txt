### ============================================================
### 弹幕战系统 — 主动挑战（玩家入口；对决战亦可由事件/接口触发）
### 入口: th_danmaku_duel_start_effect = { challenger = scope:actor target = scope:target }
### 选择流：第 1 步选幻想乡 IO → 第 2 步选对手成员（interaction_source_list 过滤：
### 自己与对决中者不出现；对齐 th_gensokyo_request_expansion 两步选择模式）
### ============================================================

th_danmaku_challenge = {
    type = internationalorganization

    ai_tick = monthly
    ai_tick_frequency = 3

    potential = {
        scope:actor = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
            NOT = {
                th_danmaku_duel_in_progress_trigger = yes
            }
        }
    }

    allow = {
        always = yes
    }

    ### 第 1 步：选幻想乡 IO（对齐 th_gensokyo_request_expansion 模式；
    ### visible 按 IO 类型过滤，玩家仅可见/可选幻想乡 IO）
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = {
            data = name
        }
        visible = {
            international_organization_type = international_organization_type:th_gensokyo_io
        }
    }

    ### 第 2 步：选对手成员（interaction_source_list = 幻想乡 IO 全部成员，
    ### limit 排除自己与对决中者；列 = 国名）
    select_trigger = {
        looking_for_a = country
        source = recipient
        target_flag = target
        name = "choose_country"
        none_available_msg_key = "th_danmaku_challenge_none_available"
        column = {
            data = name
        }
        interaction_source_list = {
            scope:recipient = {
                every_international_organization_member = {
                    limit = {
                        NOT = {
                            this = scope:actor
                        }
                        NOT = {
                            th_danmaku_duel_in_progress_trigger = yes
                        }
                    }
                    add_to_list = source
                }
            }
        }
        visible = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        enabled = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }

    effect = {
        ### 2026-08-26 同款守卫：AI 预评估/tooltip 未选 target 时直接引用报
        ### "Undefined event target"（error.log 实证教训，同 request_expansion B1 修复）
        if = {
            limit = {
                exists = scope:target
            }
            ### 2026-08-28 修复：行动 effect 顶层隐式 scope 为空（P1-12 教训），
            ### 效果内一律不使用隐式 ROOT —— 显式传发起方 challenger = scope:actor
            ### （玩家用行动时即 player），对手 target = scope:target（select_trigger 保存，
            ### 在 effect 顶层解析最可靠）。select_trigger 作用域与 scope:actor 均在顶层求值后
            ### 文本替换进 th_danmaku_duel_start_effect，不再依赖任何隐式/嵌套作用域。
            th_danmaku_duel_start_effect = {
                challenger = scope:actor
                target = scope:target
            }
        }
    }

    cooldown = {
        type = th_danmaku_challenge
        ### 弹幕挑战冷却：5 年（60 个月）
        months = 60
    }

    ai_will_do = {
        ### AI 不主动使用本行动（AI 对决由事件/异变接口驱动）
        add = {
            desc = BASE_VALUE
            value = 0
        }
    }
}
