### ============================================================
### 审判庭 — 升档（T2-2b 新增，2026-10-01）
### 规格：《脚本规格》§4.2、《设计定案》§5.3 升级门槛 + §5.4 代理人选取
###
### 与相邻行动的边界：
###   th_gensokyo_tribunal_indict（T2-2）  = 立案，施加第 1 档
###   本行动                                = 观察期届满未悔改 → 第 N 档 → 第 N+1 档
###   th_gensokyo_crusade_declare（T2-4 改用）= 走完第 4 档后开庭
###
### 【时序总纲 —— 本窗最关键的一条约束】
###   `allow` 只要求"存在被诉国且观察期已归零"（**正向条件**，
###   不要照抄《脚本规格》§4.2 表里写成 NOT —— 那是反的。
###   照 th_gensokyo_tribunal_indict.txt:52-60 的既有口径：
###   custom_tooltip 的文案写"阻塞状态"，内层写正向条件）。
###   点击后：**档位不落地**，只提名 + 发同意事件；
###   被提名国接受 → 档位才 +1；拒绝 → 保留原档位（"该档暂挂"）。
###   落地逻辑全部在 scripted_effects/th_gensokyo_tribunal_effects.txt 的
###   th_gensokyo_tribunal_escalate_land_effect（含完整时序说明）。
### ============================================================

th_gensokyo_tribunal_escalate = {
    type = internationalorganization

    ai_tick = monthly
    ai_tick_frequency = 3

    potential = {
        ### P1-12 纪律：potential 顶层 root 为空，必须显式 scope:actor 包装
        ### （蓝本：本文件的行动姊妹 generic_actions/th_gensokyo_tribunal_indict.txt:29-45）
        scope:actor ?= {
            ### 只认阎魔（幻想乡 IO 领袖）；三贤者无升档权
            is_leader_of_international_organization = international_organization:th_gensokyo_io
        }
    }

    allow = {
        ### 铁律 3：any_country + scope 依赖值必须用 custom_tooltip 包住
        ### （jomini_trigger_description.cpp:205 刷屏实证；
        ###  蓝本 generic_actions/th_gensokyo_crusade_declare_war.txt:34-54：
        ###  tooltip 文案写"阻塞状态"、内层写**正向**条件）
        custom_tooltip = {
            text = TH_TRIBUNAL_NO_ESCALATION_TT
            any_country = {
                ### 有被诉国（案件判据恒为 tarnish 标记，不是 observation_months）
                has_variable = th_gensokyo_tarnish_mark
                ### 观察期已归零（由月度脉冲的悔改判定打出）
                has_variable = th_gensokyo_observation_done
                ### 第 4 档已是阶梯顶点：再往上走是"开庭"（§4.7），不是升档
                NOT = { has_variable = th_gensokyo_tier_4_mark }
            }
        }
    }

    ### ------------------------------------------------------------
    ### ① 选幻想乡 IO（recipient 注入，供 ②③ 的 interaction_source_list 使用）
    ### 逐字照 th_gensokyo_tribunal_indict.txt:65-77
    ### ------------------------------------------------------------
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        ### 铁律 4：每个 select_trigger 必须带 column = { data = name } 防 assertion
        column = {
            data = name
        }
        visible = {
            international_organization_type = international_organization_type:th_gensokyo_io
        }
    }

    ### ------------------------------------------------------------
    ### ② 选被诉国：幻想乡 IO 成员中持「祟」且观察期已归零者
    ### 照 th_gensokyo_tribunal_indict.txt:81-126
    ### ------------------------------------------------------------
    select_trigger = {
        looking_for_a = country
        source = actor
        target_flag = target_country
        name = "choose_country"
        column = {
            data = name
        }
        column = {
            data = population
        }
        column = {
            data = great_power_score
        }
        interaction_source_list = {
            ### 铁律 16：候选列表用 interaction_source_list + add_to_list = source
            scope:recipient = {
                every_international_organization_member = {
                    limit = {
                        has_variable = th_gensokyo_tarnish_mark
                        has_variable = th_gensokyo_observation_done
                        NOT = { has_variable = th_gensokyo_tier_4_mark }
                    }
                    add_to_list = source
                }
            }
        }
        visible = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        enabled = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }

    ### ------------------------------------------------------------
    ### ③ 代理人点名（本窗重点）
    ### 候选池 = 《脚本规格》§5 的七条合格候选，逐条落进 limit
    ###   ① 幻想乡 IO 成员 —— 由 every_international_organization_member 天然限定
    ###   ② 非被诉国
    ###   ③ 非被诉国盟友 / 附属 / 宗主（写法照
    ###      generic_actions/th_gensokyo_crusade_declare.txt:201-216）
    ###   ④ 非免制裁期
    ###   ⑤ 非彼岸监管国（裸键，不带 subject_type: 前缀 —— 全库 258 处
    ###      裸键 / 0 处前缀，见 country_interactions 到 laws 的分布；
    ###      本项目同款 generic_actions/th_gensokyo_crusade_declare.txt:223）
    ###   ⑥ 非曾拒绝领衔
    ###   ⑦ 硬门槛：相对实力 >= 1.0（不弱于被诉国）
    ### 排序：《脚本规格》§5 的"距离最近"优先级 1-3 无逐字蓝本
    ###   （R3 §R3-1 列为实机验证项）⇒ 本窗不启用排序，
    ###   候选池交给引擎默认顺序；玩家可自行挑（点名本就是玩家的选择）。
    ### ------------------------------------------------------------
    select_trigger = {
        looking_for_a = country
        source = actor
        target_flag = agent_country
        ### 第二个"选国家"阶段用 vanilla 已有的另一个阶段标题键
        ### （VANILLA\common\generic_actions\golden_age_of_piracy.txt:189 用到；
        ###  避免同一行动内两个 select_trigger 用同一个 name）
        name = "choose_a_country"
        column = {
            data = name
        }
        column = {
            data = population
        }
        column = {
            data = great_power_score
        }
        interaction_source_list = {
            scope:recipient = {
                every_international_organization_member = {
                    limit = {
                        ### ② 非被诉国
                        NOT = { has_variable = th_gensokyo_tarnish_mark }
                        ### ③a 非被诉国的盟友
                        NOT = {
                            is_allied_with = {
                                target = scope:target_country
                            }
                        }
                        ### ③b 非被诉国的附属
                        NOT = { is_subject_of = scope:target_country }
                        ### ③c 非被诉国的宗主
                        NOT = {
                            any_subject = {
                                this = scope:target_country
                            }
                        }
                        ### ④ 非免制裁期
                        NOT = { has_variable = th_gensokyo_sanction_immunity }
                        ### ⑤ 非彼岸监管国
                        NOT = { is_subject_type = th_gensokyo_regulated_state }
                        ### ⑥ 非曾拒绝领衔
                        NOT = { has_variable = th_gensokyo_agent_refused }
                        ### ⑦ 硬门槛：不弱于被诉国（否则领衔者打不赢）
                        relative_strength = {
                            target = scope:target_country
                            value >= 1.0
                        }
                    }
                    add_to_list = source
                }
            }
        }
        visible = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        enabled = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }

    effect = {
        ### 铁律 2：exists 守卫防 tooltip 预演报错
        ### （蓝本 generic_actions/th_gensokyo_tribunal_indict.txt:156-159；
        ###  vanilla lordship_of_ireland.txt:193-216）
        if = {
            limit = {
                exists = scope:target_country
                exists = scope:agent_country
            }
            ### ① 清旧代理人标记（照 T1 写法：同一标记全局唯一，
            ###    否则上一任代理人会在关系图 / 圣战 AI 权重里残留）
            ###    蓝本 generic_actions/th_gensokyo_crusade_declare.txt:128-133
            ###    顺带清掉可能残留的**暂挂提名标记**（上一轮升档若因案件被
            ###    结案而未走到接受/拒绝，该标记不会自行消失）
            every_country = {
                limit = {
                    OR = {
                        has_variable = th_gensokyo_tribunal_agent_mark
                        has_variable = th_gensokyo_tribunal_agent_nominee_mark
                    }
                }
                if = {
                    limit = { has_variable = th_gensokyo_tribunal_agent_mark }
                    remove_variable = th_gensokyo_tribunal_agent_mark
                }
                if = {
                    limit = { has_variable = th_gensokyo_tribunal_agent_nominee_mark }
                    remove_variable = th_gensokyo_tribunal_agent_nominee_mark
                }
            }
            ### ② 提名：暂挂标记（征求意见期间的身份）+ 临时领衔标记。
            ###    **档位不在这里落地** —— 落地要等被提名国接受
            ###    （见 scripted_effects 的 th_gensokyo_tribunal_escalate_land_effect 头注）
            scope:agent_country = {
                set_variable = th_gensokyo_tribunal_agent_nominee_mark
                set_variable = th_gensokyo_tribunal_agent_mark
                ### ③ 征求同意（被提名国弹出事件；拒绝即保留原档位）
                ###    通知手段蓝本：scripted_effects/th_gensokyo_expansion_effects.txt:131-134、
                ###    generic_actions/th_gensokyo_request_expansion.txt:226-231
                ###    （trigger_event_non_silently = { id = <namespace>.<n> days = 1 }）
                trigger_event_non_silently = {
                    id = th_gensokyo_tribunal_events.1
                    days = 1
                }
            }
        }
    }

    ### 铁律 12：冷却写法 = { type = <key> months = N }
    ### 字段定义 VANILLA\common\generic_actions\readme.txt:73
    ### "cooldown = { type = <any tag> days/weeks/months/years = <integer> }"
    cooldown = {
        type = th_gensokyo_tribunal_escalate
        months = 3
    }

    ai_will_do = {
        ### 铁律 5 的 v1.8 修订版：ai_will_do **可以**用 has_variable 与 var:，
        ### 前提是变量先存在（正证 VANILLA\common\generic_actions\io_parliament_bribes.txt:235/251/253）
        ### 铁律 6：不用 any_international_organization_member（scope 不匹配实测报
        ### "Inconsistent trigger scopes"），改 any_country + 显式成员判定
        if = {
            limit = {
                any_country = {
                    is_member_of_international_organization = international_organization:th_gensokyo_io
                    has_variable = th_gensokyo_tarnish_mark
                    has_variable = th_gensokyo_observation_done
                    NOT = { has_variable = th_gensokyo_tier_4_mark }
                }
            }
            add = {
                desc = TH_TRIBUNAL_AI_ESCALATE_READY
                value = 500
            }
        }
        add = {
            desc = BASE_VALUE
            value = 1
        }
    }
}
