th_gensokyo_propose_crusade = {
    type = internationalorganization
    
    ai_tick = monthly
    ai_tick_frequency = 6
    
    potential = {
        ### P1-12 修复：potential 顶层 root=empty，is_leader 裸挂恒 false → 提案按钮不显示
        scope:actor = {
            OR = {
                has_special_status_in_international_organization = {
                    type = special_status:th_gensokyo_io_sages_special_status
                    international_organization = international_organization:th_gensokyo_io
                }
                is_leader_of_international_organization = international_organization:th_gensokyo_io
            }
        }
    }
    
    allow = {
        international_organization:th_gensokyo_io = {
            NOT = {
                is_active_parliament = yes
            }
            NOT = {
                situation:th_gensokyo_crusade = {
                    situation_is_active = yes
                }
            }
        }
        ### 2026-08-21（q7 决策）：辩论失败后领袖 12 月冷却（on_debate_failed 打标）
        NOT = {
            has_cooldown = th_gensokyo_propose_crusade
        }
    }
    
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = {
            data = name
        }
        visible = {
            international_organization_type = international_organization_type:th_gensokyo_io
        }
    }
    
    select_trigger = {
        looking_for_a = country
        target_flag = target_country
        ### B1 修复 2026-08-14：name 是引擎内置选择器模板 ID（base game 无任何行动用自造名），
        ### "choose_crusade_target" 导致选国 UI 不绑定 target_flag → 1179 次
        ### "Undefined event target"；改用内置 "choose_country"（japanese_shogunate.txt:64 先例），
        ### target_flag 对齐 high_kingship_of_ireland.txt:171 的 target_country
        name = "choose_country"
        ### 2026-08-23 实测修复：补 column——原缺 column 导致 select-interaction-target
        ### "There is no column to display" assertion（16:41:52 实证）→ 选择流程失败 →
        ### 目标/recipient 注入异常 → call_io_parliament "Scope: empty"（16:41:53 实证）。
        ### 对齐先例：japanese_shogunate.txt:66-68（choose_country + column）/
        ### high_kingship_of_ireland.txt:172-175（三列）
        column = {
            data = name
        }
        ### 乱价值观过滤（2026-08-19 用户决策：惩戒目标 = 缘与乱偏移 ≥ +50，
        ### 即弃民二档阈值，用 outcast 标记镜像过滤）
        ### modifier mirror 方案（2026-08-14 实测修复）：interaction_source_list 的 limit
        ### 上下文不吃 var: 数值比较（root.var: 报 "Invalid left side during comparison
        ### 'var'" + "Event target link 'root' returned an unset scope"，1756 次/秒刷屏），
        ### 改用月度脉冲落好的标记修正过滤（普通触发器，无 var）
        interaction_source_list = {
            scope:recipient = {
                every_international_organization_member = {
                    limit = {
                        NOT = {
                            this = scope:actor
                        }
                        NOT = {
                            is_leader_of_international_organization = international_organization:th_gensokyo_io
                        }
                        societal_value:th_gensokyo_amity_vs_belligerence >= th_notorious_outcast_threshold_value
                        # has_country_modifier = th_gensokyo_notorious_outcast_modifier
                        ### WP-H 2026-08-15（设计§6）：免制裁期（20 年）内不可再被惩戒
                        NOT = {
                            has_variable = th_gensokyo_sanction_immunity
                        }
                    }
                    add_to_list = source
                }
            }
        }
        visible = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        enabled = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }
    
    effect = {
        ### 清除旧标记，给目标国打新标记
        scope:recipient = {
            every_international_organization_member = {
                remove_variable = th_gensokyo_crusade_target_mark
            }
        }
        ### B1 修复：target_flag 改名 target_country + exists 守卫（防未选择时误执行）
        if = {
            limit = {
                exists = scope:target_country
            }
            scope:target_country = {
                set_variable = th_gensokyo_crusade_target_mark
            }
        }
        ### 内联发起议会辩论（贤者议会同款，root = 执行国）
        ### 2026-08-21（bug 2 用户拍板）：议会开始时所有国家（含发起人阎魔）均尚未投票，
        ### 不再全员预投 no、发起人也不再自动 vote=yes；
        ### 初始支持度 = 0，票由规则自动投票 + 成员手动/AI 投票产生
        ### 投票权重 = 幻想乡影响力（special_status_power 动态公式，见席位文件）
        scope:recipient = {
            set_parliament_location = scope:actor.capital
            set_parliament_active = yes
            set_parliament_issue = parliament_issue:th_gensokyo_crusade_issue
        }
        call_io_parliament = {
            international_organization = scope:recipient
        }
        ### 2026-08-21（实测修复）：规则自动投票必须在议会激活后执行——
        ### on_debate_start 回调触发时机早于 call_io_parliament（议会未激活），
        ### participate_in_parliament 被引擎忽略（三贤者 UI 仍"尚未投票"）；
        ### on_io_parliament_started 钩子不触发（base game 0 实例）。
        ### 本处 effect 顺序执行：call_io_parliament 返回后议会已激活，立即规则投票
        scope:recipient = {
            th_gensokyo_crusade_debate_start_effect = yes
        }
    }
    
    force_click_and_confirm_or_hold = yes
    
    ai_will_do = {
        ### B2 修复 2026-08-14：AI 提案倾向同步改恶名驱动（与选国过滤一致，用户决策）
        ### 2026-08-21（需求3）：弃民标记已删（outcast modifier 定义已移除），
        ### 改直接 societal_value 检测（字面值 50；阈值与过滤行
        ### th_notorious_outcast_threshold_value 一致，改动需同步）
        ### 2026-08-21（问题 2 用户决策）：有可惩戒目标时倾向拉满（+10000，
        ### base game "ANY vote" 同款量级）→ AI 阎魔有目标即提案
        if = {
            limit = {
                any_international_organization_member = {
                    NOT = {
                        this = scope:actor
                    }
                    NOT = {
                        is_leader_of_international_organization = international_organization:th_gensokyo_io
                    }
                    societal_value:th_gensokyo_amity_vs_belligerence >= 50
                    ### WP-H：免制裁期内不计为可惩戒目标
                    NOT = {
                        has_variable = th_gensokyo_sanction_immunity
                    }
                }
            }
            add = {
                desc = TH_CRUSADE_AI_FOUND_TARGET
                value = 10000
            }
        }
        add = {
            desc = BASE_VALUE
            value = 1
        }
    }
}
