### 设计文档《国际组织》§6：所有参与战争的国家可以投票选择自己支持的裁决判决
### （与投票无关）→ 风味投票：仅记录票数供 GUI 展示，

th_gensokyo_trial_vote = {
    type = situation

    ai_tick = monthly
    ai_tick_frequency = 6

    potential = {
        scope:actor ?= {
            is_member_of_international_organization = international_organization:th_gensokyo_io
            ### 每人每场审判限投一次
            NOT = {
                has_variable = th_gensokyo_trial_voted
            }
            ### 目标国不可自投
            NOT = {
                has_variable = th_gensokyo_trial_target_mark
            }
        }
    }

    select_trigger = {
        looking_for_a = situation
        target_flag = recipient
        name = "choose_situation"
        column = {
            data = name
        }
        visible = {
            situation:th_gensokyo_trial = this
            situation_is_active = yes
        }
    }

    effect = {
        ### 弹出投票事件（3 选项：从宽/适中/从严）
        scope:actor = {
            trigger_event_non_silently = {
                id = th_gensokyo_trial_vote_events.1
                days = 1
            }
        }
    }

    ai_will_do = {
        value = 0
    }
}
