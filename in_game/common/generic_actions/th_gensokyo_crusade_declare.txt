th_gensokyo_crusade_declare = {
    type = internationalorganization

    ai_tick = monthly
    ai_tick_frequency = 6

    ### ============================================================
    ### [2026-10-31 用户要求] 暂时禁用「建立审判庭」行动
    ### 禁用方式：potential 恒假 -> 玩家看不到按钮，AI 也不会评估。
    ### 恢复方式：删掉下面这个 potential，再把下方「原 potential（注释留档）」
    ###           的 19 行取消 ### 前缀即可。
    ### 注：本行动仍登记在 generic_action_ai_lists/th_gensokyo_io_list.txt:25，
    ###     恢复 potential 后 AI 会自动重新评估，无需改列表。
    ### ============================================================
    potential = {
        always = no
    }

    ### ---- 原 potential（注释留档，恢复时用） ----
###     potential = {
###         ### P1-12 修复：potential 顶层 root=empty，须显式 scope:actor
###         scope:actor = {
###             ### 阎魔（幻想乡 IO 领袖）——2026-10-01 T1-1：本行动发起权收窄为
###             ### 只认阎魔，三贤者分支已删（《实施计划》§1.2、《脚本规格》4.7）。
###             ### 三贤者的戏份改由"强制入庭"承载（见 effect）；代理人领衔另用
###             ### th_gensokyo_tribunal_agent_mark 标记，不代表发起权。
###             is_leader_of_international_organization = international_organization:th_gensokyo_io
###             ### 审判庭已存在时不可再组建（2026-08-26 用户需求：唯一审判庭）
###             ### 2026-08-26：exists 链接对 unique 动态 IO 不可靠 → 改
###             ### any_international_organization 遍历判定（vanilla 实证触发器；
###             ### destroy 后遍历自然失效）
###             NOT = {
###                 any_international_organization = {
###                     international_organization_type = international_organization_type:th_gensokyo_tribunal
###                 }
###             }
###         }
###     }

    allow = {
        always = yes
    }

    ### ① 选 IO（recipient 注入，供 ② 的 interaction_source_list 使用）
    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = {
            data = name
        }
        visible = {
            international_organization_type = international_organization_type:th_gensokyo_io
        }
    }

    ### ② 选目标国（好战≥50，非领袖，免制裁排除）
    select_trigger = {
        looking_for_a = country
        source = actor
        target_flag = target_country
        name = "choose_country"
        ### 2026-08-23：column 补全（选择器必备，防 assertion）
        column = {
            data = name
        }
        column = {
            data = population
        }
        column = {
            data = great_power_score
        }
        interaction_source_list = {
            scope:recipient = {
                every_international_organization_member = {
                    limit = {
                        NOT = {
                            this = scope:actor
                        }
                        NOT = {
                            is_leader_of_international_organization = international_organization:th_gensokyo_io
                        }
                        ### 好战≥50（惩戒门槛）
                        societal_value:th_gensokyo_amity_vs_belligerence >= th_notorious_outcast_threshold_value
                        ### 免制裁期（反杀后 20 年）内不可再被惩戒
                        NOT = {
                            has_variable = th_gensokyo_sanction_immunity
                        }
                    }
                    add_to_list = source
                }
            }
        }
        visible = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
        enabled = {
            is_member_of_international_organization = international_organization:th_gensokyo_io
        }
    }

    effect = {
        ### 清理旧目标标记，打新标记（审判/局势/宣战行动引用）
        every_country = {
            limit = {
                has_variable = th_gensokyo_crusade_target_mark
            }
            remove_variable = th_gensokyo_crusade_target_mark
        }
        ### 2026-10-01 T1-1：上一届代理人标记一并清掉——同一标记必须全局唯一，
        ### 否则已解散审判庭的旧代理人会在关系图 / 圣战 AI 权重里残留
        every_country = {
            limit = {
                has_variable = th_gensokyo_tribunal_agent_mark
            }
            remove_variable = th_gensokyo_tribunal_agent_mark
        }
        ### 2026-08-26 修复：exists 守卫防 tooltip 预演/选择器异常时报错
        ### （vanilla lordship_of_ireland.txt:193-216 先例；标记设置失败时
        ### 不再刷屏，create 照常执行）
        if = {
            limit = {
                exists = scope:target_country
            }
            scope:target_country = {
                set_variable = th_gensokyo_crusade_target_mark
            }
        }
        ### 创建审判庭（unique 全局唯一；2026-08-26 照 italian_wars 官方写法：
        ### 无 target/creator 参数——引擎不识别；显式 scope:actor 加入并设领袖）
        create_international_organization = {
            type = international_organization_type:th_gensokyo_tribunal
            add_country_to_international_organization = scope:actor
            set_leader_country = scope:actor
            save_scope_as = tribunal
        }
        ### 2026-08-26（用户需求）：创建者自身附庸自动加入审判庭
        ### （add_country_to_international_organization = IO scope 效果，
        ### readme.txt:165；目标国标记已先打上，排除目标本身）
        scope:actor = {
            every_subject = {
                limit = {
                    NOT = {
                        has_variable = th_gensokyo_crusade_target_mark
                    }
                }
                scope:tribunal = {
                    add_country_to_international_organization = prev
                }
            }
        }
        ### ============================================================
        ### 2026-10-01 T1-1：选世俗代理人 → 打国家标记（本窗核心）
        ### 规格：《设计定案》§5.4、《脚本规格》§5、《实施计划》§2.1 的 1-1
        ### 候选池（逐条写进 limit；成员身份由 ordered_international_organization_member
        ### 天然限定为幻想乡 IO 成员）：
        ###   ② 非被诉国 ③ 非被诉国的盟友 / 附属 / 宗主
        ###   ④ 非免制裁期 ⑤ 非彼岸监管国
        ###   ⑥ 硬门槛：相对实力 >= 1.0（不弱于被诉国，否则圣战变笑话）
        ###   （「非曾拒绝领衔」属 T2-4 同意流程，本窗不写）
        ### 排序：本窗取退路「合格候选中实力最强者」——
        ###   order_by = military_strength 有 vanilla 逐字先例
        ###   （common/situations/war_of_religions.txt:54）；
        ###   「距离最近」需 order_by = "border_distance_to(...)"，蓝本无逐字先例
        ###   （《待验证结论_R3》§R3-1 不确定 #6）→ 列为实机验证项，本窗不启用。
        ### 兜底链：合格候选为空 → 三贤者中实力最强者（贤者必然参战，永不为空）
        ### 写入位置：国家标记（审判庭 IO 无法持久保存 scope，《脚本规格》§5 :282）
        ### ============================================================
        if = {
            limit = {
                exists = scope:target_country
            }
            international_organization:th_gensokyo_io = {
                ordered_international_organization_member = {
                    limit = {
                        ### ① 排除阎魔自己（《设计定案》§3.1「彼岸不动手」：法是彼岸给的，
                        ### 动手的是世俗代理人）。写法照本文件 select_trigger ② 的既有守卫
                        NOT = {
                            this = scope:actor
                        }
                        ### ② 非被诉国
                        NOT = {
                            has_variable = th_gensokyo_crusade_target_mark
                        }
                        ### ③a 非被诉国的盟友
                        NOT = {
                            is_allied_with = {
                                target = scope:target_country
                            }
                        }
                        ### ③b 非被诉国的附属（本国是被告的属国）
                        NOT = {
                            is_subject_of = scope:target_country
                        }
                        ### ③c 非被诉国的宗主（被告是本国的属国）
                        NOT = {
                            any_subject = {
                                this = scope:target_country
                            }
                        }
                        ### ④ 非免制裁期
                        NOT = {
                            has_variable = th_gensokyo_sanction_immunity
                        }
                        ### ⑤ 非彼岸监管国
                        NOT = {
                            is_subject_type = th_gensokyo_regulated_state
                        }
                        ### ⑥ 硬门槛：不弱于被诉国
                        relative_strength = {
                            target = scope:target_country
                            value >= 1.0
                        }
                    }
                    ### 排序键 = 绝对军力；max = 1 取该维度的第一名
                    order_by = military_strength
                    max = 1
                    check_range_bounds = no
                    save_scope_as = tribunal_agent
                }
            }
            ### 兜底：合格候选一个都没有 → 三贤者中实力最强者
            if = {
                limit = {
                    NOT = {
                        exists = scope:tribunal_agent
                    }
                }
                international_organization:th_gensokyo_io = {
                    ordered_international_organization_member = {
                        limit = {
                            has_special_status_in_international_organization = {
                                type = special_status:th_gensokyo_io_sages_special_status
                                international_organization = international_organization:th_gensokyo_io
                            }
                        }
                        order_by = military_strength
                        max = 1
                        check_range_bounds = no
                        save_scope_as = tribunal_agent
                    }
                }
            }
            ### 打标记。最终仍选不出人（理论上不该发生）时整块跳过，
            ### 建庭流程照常走完，不报错、不中断
            if = {
                limit = {
                    exists = scope:tribunal_agent
                }
                scope:tribunal_agent = {
                    set_variable = th_gensokyo_tribunal_agent_mark
                }
            }
        }
        ### 代理人强制入庭（《设计定案》§5.4 第 3 条：硬加，绕过 can_join_trigger）
        if = {
            limit = {
                exists = scope:tribunal_agent
            }
            scope:tribunal = {
                add_country_to_international_organization = scope:tribunal_agent
            }
        }
        ### 三贤者强制入庭（special_status 判定照本文件原 :36-39 的写法，不硬编码 tag）。
        ### 去重：以 th_gensokyo_tribunal 成员身份过滤——代理人若本身即贤者，
        ### 上一步已入庭，此处不会重复添加
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                limit = {
                    has_special_status_in_international_organization = {
                        type = special_status:th_gensokyo_io_sages_special_status
                        international_organization = international_organization:th_gensokyo_io
                    }
                    NOT = {
                        is_member_of_international_organization = international_organization:th_gensokyo_tribunal
                    }
                }
                scope:tribunal = {
                    add_country_to_international_organization = prev
                }
            }
        }
        ### 2026-08-23（用户修正）：不在本行动宣战——
        ### 等待各国加入审判庭后，用 th_gensokyo_crusade_declare_war 行动宣战；
        ### 宣战后 on_war_declared 钩子召唤全体成员入战（局势已删除）
    }

    ai_will_do = {
        ### 有可惩戒目标（好战≥50）且 actor 不弱于目标 → 发起审判庭
        ### 2026-08-26：any_international_organization_member 在 ai_will_do 上下文
        ### scope 不匹配（"Inconsistent trigger scopes" 实证）→ 改
        ### any_country + is_member 显式判定（同 join_crusade 已验证写法）
        if = {
            limit = {
                any_country = {
                    is_member_of_international_organization = international_organization:th_gensokyo_io
                    NOT = {
                        this = scope:actor
                    }
                    NOT = {
                        is_leader_of_international_organization = international_organization:th_gensokyo_io
                    }
                    societal_value:th_gensokyo_amity_vs_belligerence >= th_notorious_outcast_threshold_value
                    NOT = {
                        has_variable = th_gensokyo_sanction_immunity
                    }
                    scope:actor = {
                        relative_strength = {
                            target = prev
                            value >= 0.5
                        }
                    }
                }
            }
            add = {
                desc = TH_TRIBUNAL_AI_TARGET_FOUND
                value = 10000
            }
        }
        add = {
            desc = BASE_VALUE
            value = 1
        }
    }
