th_gensokyo_trial_issue = {
    type = international_organization

    potential = {
        always = yes
    }
    allow = {
        always = yes
    }
    selectable_for = {
        is_leader_of_international_organization = scope:recipient
    }
    chance = 100

    on_debate_passed = {
        ### ============ 从重意向（通过 = 多数支持从重）============
        ### 只写意向，**不再判**。IO 变量 +1（审判庭此刻必然存在 ——
        ### 本议题就是它的议会发起的）。
        ### 写法：IO scope 块 set_variable + 负数字面量，蓝本
        ###   VANILLA\common\scripted_effects\situation_effects.txt:618
        ###   （set_variable = { name = … value = -1 }）
        ###   IO 块形式跳转蓝本：international_organization_effects.txt:1033
        international_organization:th_gensokyo_tribunal = {
            set_variable = {
                name = th_gensokyo_verdict_intent
                value = 1
            }
        }
        ### 国家标记：从重（两意向标记互斥，本次辩论只可能是其中之一；
        ### 仍显式摘掉另一个，防上一案残留 —— 与宣判路径的清法一致）
        every_country = {
            limit = {
                has_variable = th_gensokyo_crusade_target_mark
            }
            set_variable = th_gensokyo_intent_harsh
            if = {
                limit = { has_variable = th_gensokyo_intent_soft }
                remove_variable = th_gensokyo_intent_soft
            }
        }
        ### ============================================================
        ### 2026-10-04 闭环改造④：辩论结束 ⇒ 彼岸（审判庭领袖）跳裁决事件
        ###   scope 链 IO → 领袖国：项目内既有用法
        ###     on_action\th_gensokyo_chain_events.txt:719-722
        ###       （international_organization:… = { leader_country = { … } }）
        ###   触发 effect 蓝本（★ EU5 **没有** trigger_event，只有 silently/non_silently）：
        ###     VANILLA\events\character\court_flavor_events.txt:112
        ###       trigger_event_non_silently = { id = court_flavor_events.3 days = 15 }
        ###   语义：辩论一出结果就把裁决权交到阎魔手上，不必再等阎魔去点面板
        ###   超时兜底保留（th_gensokyo_verdict_overdue）：阎魔不点则按意向自动落档
        ### ============================================================
        international_organization:th_gensokyo_tribunal = {
            leader_country ?= {
                trigger_event_non_silently = {
                    id = th_gensokyo_tribunal_events.4
                    days = 1
                }
            }
        }
    }

    on_debate_failed = {
        ### ============ 从轻意向（失败 = 从重未获多数）============
        international_organization:th_gensokyo_tribunal = {
            set_variable = {
                name = th_gensokyo_verdict_intent
                value = -1
            }
        }
        every_country = {
            limit = {
                has_variable = th_gensokyo_crusade_target_mark
            }
            set_variable = th_gensokyo_intent_soft
            if = {
                limit = { has_variable = th_gensokyo_intent_harsh }
                remove_variable = th_gensokyo_intent_harsh
            }
        }
        ### 2026-10-04 闭环改造④：辩论否决（= 从轻意向）同样立即交裁决权（注释见 on_debate_passed）
        international_organization:th_gensokyo_tribunal = {
            leader_country ?= {
                trigger_event_non_silently = {
                    id = th_gensokyo_tribunal_events.4
                    days = 1
                }
            }
        }
    }

    ### AI 投票倾向（站队二分；root = 投票国，scope:recipient = 幻想乡 IO）
    ### 2026-08-23：审判无复杂倾向（用户决策：倾向不好写，放弃精细模型），
    ### 只按站队二分——参战/交战者从重，目标方从轻，其余中立按基础值
    wants_this_parliament_issue_bias = {
        ### 参战成员（审判庭成员/participant 标记）→ 从重
        if = {
            limit = {
                OR = {
                    has_variable = th_gensokyo_crusade_participant
                    is_member_of_international_organization = international_organization:th_gensokyo_tribunal
                }
            }
            add = {
                desc = TH_TRIAL_BIAS_PARTICIPANT
                value = 500
            }
        }
        ### 与目标交战 → 从重
        if = {
            limit = {
                any_country = {
                    has_variable = th_gensokyo_crusade_target_mark
                    root = {
                        is_at_war_with = prev
                    }
                }
            }
            add = {
                desc = TH_TRIAL_BIAS_AT_WAR
                value = 300
            }
        }
        ### 目标盟友/附属 → 从轻
        if = {
            limit = {
                any_country = {
                    has_variable = th_gensokyo_crusade_target_mark
                    OR = {
                        root = {
                            is_allied_with = {
                                target = prev
                            }
                        }
                        root = {
                            is_subject_of = prev
                        }
                        prev = {
                            is_subject_of = root
                        }
                    }
                }
            }
            add = {
                desc = TH_TRIAL_BIAS_TARGET_SIDE
                value = -500
            }
        }
        ### 基础值：略偏从轻（默认宽容，参战方推动才从重）
        add = {
            desc = "VOTE_BASE_VALUE"
            value = -20
        }
    }
}
