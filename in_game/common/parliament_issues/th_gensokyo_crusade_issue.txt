th_gensokyo_crusade_issue = {

    special_status = th_gensokyo_io_yama_special_status

    potential = {
        always = yes
    }
    allow = {
        always = yes
    }
    selectable_for = {
        is_leader_of_international_organization = scope:recipient
    }
    chance = 100

    ### 规则自动投票已移至 propose_crusade effect（call_io_parliament 之后、议会激活后执行——
    ### 2026-08-21 实测：on_debate_start 回调触发时机早于议会激活，投票被引擎忽略；
    ### on_io_parliament_started 钩子不触发，均已移除）

    on_debate_passed = {
        international_organization:th_gensokyo_io = {
            ### 2026-08-23 实测修复：enable_situation_effect/activate_situation 必须
            ### 在国家 scope 调用——其 effect_localization 文本含 [COUNTRY.GetName]
            ### （THIRD_ENABLE_SITUATION_EFFECT），IO scope 渲染失败（error.log
            ### "Promote 'COUNTRY' returned nullptr" 刷屏实证）→ custom_description

            ### 挪到 leader_country（阎魔国家）scope：tooltip 正常渲染 + 效果可靠执行

            every_international_organization_member = {

                remove_variable = th_gensokyo_crusade_target_mark
                ### 2026-08-21：辩论失败不启动局势 → 同意者标记一并清除

    ### AI 投票倾向（2026-08-21，grill 拷问 2 证据链方案）：
    ### ai_parliament_issue_resolution_vote_bias（io_parliament.txt:428 决定 AI 投 yes/no）
    ### 在 parliament_issue scope 求值（wiki Trigger 页实证），脚本入口 = 本字段；
    ### 写法参照 hre_request_additional_military_support（10_hre_issues.txt:74）：
    ### 基础 -400 默认反对 + 条件加成翻正。
    ### root = 投票国，scope:recipient = IO（parliament_issues/readme.txt:13 实证）
    ### 2026-08-23 实测修复：bias 是 scripted maths 上下文（readme:13），
    ### 原 save_scope_as（effect）非法 → "Unexpected token: save_scope_as"（16:39:30
    ### 实证）→ 整个 bias 解析失败 → AI 投票倾向失效。
    ### 重写为 trigger 级定位：any_country + has_variable 找目标 + root（投票国）嵌套
    ### 判断（join_crusade potential 同款模式，实证合法）；
    ### 原「敌意≥50 动态项」改为「与目标交战者必赞成」近似（bias 无法取目标 scope
    ### 的 antagonism 数值——无 save_scope_as 则无目标 scope；P1 调参时如需精确
    ### 敌意项需另寻方案）
    wants_this_parliament_issue_bias = {
        if = {
            limit = {
                ### 存在惩戒目标（标记法）
                any_country = {
                    has_variable = th_gensokyo_crusade_target_mark
                }
            }
            add = {
                desc = "VOTE_BASE_VALUE"
                value = -400
            }
            ### 与目标交战者：必赞成（原敌意≥50 动态项的 trigger 级近似）
            if = {
                limit = {
                    any_country = {
                        has_variable = th_gensokyo_crusade_target_mark
                        root = {
                            is_at_war_with = prev
                        }
                    }
                }
                add = {
                    desc = TH_CRUSADE_AI_FOUND_TARGET
                    value = 500
                }
            }
            ### 贤者：必赞成
            if = {
                limit = {
                    th_gensokyo_is_sage_trigger = yes
                }
                add = {
                    desc = TH_CRUSADE_AI_FOUND_TARGET
                    value = 500
                }
            }
            ### 领袖（发起人阎魔）：必赞成
            if = {
                limit = {
                    is_leader_of_international_organization = scope:recipient
                }
                add = {
                    desc = TH_CRUSADE_AI_FOUND_TARGET
                    value = 500
                }
            }
            ### 目标国盟友/附属国：必反对
            if = {
                limit = {
                    any_country = {
                        has_variable = th_gensokyo_crusade_target_mark
                        OR = {
                            root = {
                                is_allied_with = {
                                    target = prev
                                }
                            }
                            root = {
                                is_subject_of = prev
                            }
                        }
                    }
                }
                add = {
                    desc = TH_CRUSADE_AI_FOUND_TARGET
                    value = -1000
                }
            }
        }
        else = {
            add = {
                desc = "VOTE_BASE_VALUE"
                value = 0
            }
        }
    }
