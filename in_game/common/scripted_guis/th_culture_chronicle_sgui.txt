th_culture_chronicle_refresh = {
	scope = country
	is_shown = {
		always = yes
	}
	is_valid = {
		always = yes
	}
	effect = {
		set_variable = { name = th_cc_page value = 0 }
		set_variable = { name = th_cc_page_start value = 0 }
		set_variable = { name = th_cc_page_end value = 5 }
		th_culture_chronicle_refresh_lists = yes
	}
	ai_is_valid = {
		always = no
	}
}

th_cc_prev_page = {
	scope = country
	is_shown = {
		always = yes
	}
	is_valid = {
		var:th_cc_page > 0
	}
	effect = {
		change_variable = { name = th_cc_page subtract = 1 }
		change_variable = { name = th_cc_page_start subtract = 5 }
		change_variable = { name = th_cc_page_end subtract = 5 }
		th_cc_refresh_display = yes
	}
	ai_is_valid = {
		always = no
	}
}

th_cc_next_page = {
	scope = country
	is_shown = {
		always = yes
	}
	is_valid = {
		var:th_cc_has_next_page = 1
	}
	effect = {
		change_variable = { name = th_cc_page add = 1 }
		change_variable = { name = th_cc_page_start add = 5 }
		change_variable = { name = th_cc_page_end add = 5 }
		th_cc_refresh_display = yes
	}
	ai_is_valid = {
		always = no
	}
}
