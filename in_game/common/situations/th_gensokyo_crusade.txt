### ============================================================
### 幻想乡惩戒局势（加强版包围网）
### 启动: 议会辩论通过（on_debate_passed → enable_situation_effect 设全局变量
###       enabled_situation_th_gensokyo_crusade → can_start 检测激活）
### 修复 2026-08-21：can_start = always no 导致 enable_situation_effect 设的
###       全局变量从不被检测 → 局势永不启动（用户实测 bug 5）
###       改为 has_enabled_situation_trigger（rise_of_timur 同款）；
###       monthly_spawn_chance 0 → 1 保证 can_start 被月度评估；
###       on_ended 加 disable_situation_effect 清全局变量（防局势结束后
###       满足条件自动重启——惩戒战争会重复发生，与一次性 rise_of_timur 不同）
### 目标国: 标记法定位（th_gensokyo_crusade_target_mark，常驻至局势结束）
### 结束: 目标国覆灭/战争全结束 → 反杀则正常结束（特殊行动解锁），否则启动审判局势
### ============================================================

th_gensokyo_crusade = {
    monthly_spawn_chance = 1

    can_start = {
        has_enabled_situation_trigger = {
            type = th_gensokyo_crusade
        }
    }
    can_end = {
        always = no
    }

    visible = {
        is_member_of_international_organization = international_organization:th_gensokyo_io
    }

    on_start = {
        ### 定位领袖（阎魔）与目标国
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                limit = {
                    is_leader_of_international_organization = international_organization:th_gensokyo_io
                }
                save_scope_as = io_leader
            }
            every_international_organization_member = {
                limit = {
                    has_variable = th_gensokyo_crusade_target_mark
                }
                save_scope_as = crusade_target
            }
        }
        ### 罪值清零
        scope:crusade_target = {
            set_variable = {
                name = th_gensokyo_sin_var
                value = 0
            }
        }
        ### 新惩戒开启：重置上一场反杀标记（反攻扩张行动随新惩戒重新锁定）
        ### 2026-08-21：同意者标记一并重置（防上一场残留影响本场加入倾向）
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                remove_variable = th_gensokyo_crusade_countered
                remove_variable = th_gensokyo_crusade_supporter
            }
        }
        ### 局势变量：目标国引用（GUI 双国头部动态头像读取）+ 罪值显示初值
        set_variable = {
            name = th_crusade_target_ref
            value = scope:crusade_target
        }
        set_variable = {
            name = th_crusade_sin_display
            value = 0
        }
        ### 大规模战争扣团结 暂未启用（B6 决策：扣团结的"大规模战争"指代范围待定）
        ### 恢复时取消注释并登记交接文档：
        # international_organization:th_gensokyo_io = {
        #     th_gensokyo_unity_subtract_effect = {
        #         value = th_unity_big_war_value
        #     }
        # }
        ### 阎魔立即宣战（决议通过即开战）
        scope:io_leader = {
            declare_war_with_cb = {
                target = scope:crusade_target
                type = casus_belli:th_gensokyo_crusade_cb
            }
        }
    }

    on_monthly = {
        ### 重新定位目标国（回调段内 scope 不跨段保留）
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                limit = {
                    has_variable = th_gensokyo_crusade_target_mark
                }
                save_scope_as = crusade_target
            }
        }
        ### 修复 2026-08-19：目标国覆灭/被吞并时 save_scope_as 不触发 →
        ### scope:crusade_target 未设置会报错；先判定目标不存在 → 直接结束局势
        if = {
            limit = {
                NOT = {
                    exists = scope:crusade_target
                }
            }
            end_situation = situation:th_gensokyo_crusade
        }
        else = {
            ### 罪值：战争时长累积（封顶 100）+ 同步到局势变量（GUI 显示）
            scope:crusade_target = {
                change_variable = {
                    name = th_gensokyo_sin_var
                    add = th_crusade_sin_monthly_value
                }
                clamp_variable = {
                    name = th_gensokyo_sin_var
                    min = 0
                    max = 100
                }
            }
            set_variable = {
                name = th_crusade_sin_display
                value = scope:crusade_target.var:th_gensokyo_sin_var
            }
        }
        ### 包围网入战由局势行动 th_gensokyo_join_crusade 触发（B2 决策）
        ### 势力计算（B5：局势面板双方力量对比，situation 变量供 GUI 读取）
        ### P1-10/P1-13 修复：两个变量统一存在局势 scope（GUI 从 SituationView 读）；
        ### 取数用 country_strength（current_strength 只是关系权重 key，非实力值）
        set_variable = {
            name = th_crusade_supporter_strength
            value = 0
        }
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                limit = {
                    has_variable = th_gensokyo_crusade_participant
                }
                root = {
                    change_variable = {
                        name = th_crusade_supporter_strength
                        add = prev.country_strength
                    }
                }
            }
        }
        set_variable = {
            name = th_crusade_target_strength
            value = 0
        }
        ### 目标国力量 + 结束判定（非覆灭路径；覆灭已在上面 else 分支外处理）
        if = {
            limit = {
                exists = scope:crusade_target
            }
            set_variable = {
                name = th_crusade_target_strength
                value = scope:crusade_target.country_strength
            }
            scope:crusade_target = {
                if = {
                    limit = {
                        at_war = no
                    }
                    end_situation = situation:th_gensokyo_crusade
                }
            }
        }
    }

    on_ending = {
        ### 反杀标记不在此清理（P1-04 修复：on_ended 需先检测 countered 再清理，
        ### 否则反杀结局会被误判为普通结局启动审判）
        ### countered 标记生命周期：反杀条约打标 → 本局势 on_ended 检测（不清理）→
        ### 反攻扩张行动持久解锁 → 下一次惩戒局势 on_start 重置
    }

    on_ended = {
        ### 2026-08-21 修复：清除 enable_situation_effect 设的全局变量
        ### （disable_situation_effect = remove_global_variable，situation_effects.txt:15），
        ### 防局势结束后 can_start 仍满足 → 下月自动重启（无需议会通过）
        ### 2026-08-23 实测修复：disable_situation_effect 的 tooltip 文本含
        ### [COUNTRY.GetName]（THIRD_DISABLE_SITUATION_EFFECT，与 enable 同源）——
        ### situation scope 渲染失败会中断 remove_global_variable → 全局变量残留 →
        ### 局势结束自动重启。挪到 leader_country（国家 scope）保证可靠执行
        international_organization:th_gensokyo_io = {
            leader_country ?= {
                disable_situation_effect = {
                    type = th_gensokyo_crusade
                }
            }
        }
        ### 非反杀结局 → 惩戒战争平息奖励团结（C2）→ 启动审判局势（target_mark 保留，审判局势 on_start 定位后自行清理）
        if = {
            limit = {
                NOT = {
                    any_country = {
                        has_variable = th_gensokyo_crusade_countered
                    }
                }
            }
            international_organization:th_gensokyo_io = {
                th_gensokyo_unity_add_effect = {
                    value = th_unity_crusade_settled_value
                }
                ### 2026-08-23 实测修复：enable_situation_effect/activate_situation 必须
                ### 在国家 scope 调用（on_ended root = situation，无 COUNTRY → tooltip
                ### 渲染失败中断效果，与 on_debate_passed 同源问题）——
                ### 挪到 leader_country（阎魔国家）scope；审判 on_start 会 disable
                ### 全局变量（单次流程自洽）
                leader_country ?= {
                    enable_situation_effect = {
                        type = th_gensokyo_trial
                    }
                    activate_situation = situation:th_gensokyo_trial
                }
            }
        }
        else = {
            ### 反杀结局：无审判，清理目标标记（countered 标记保留供反攻扩张行动使用）
            international_organization:th_gensokyo_io = {
                every_international_organization_member = {
                    remove_variable = th_gensokyo_crusade_target_mark
                }
            }
        }
        ### 清理参战标记（势力计算用）+ 同意者标记（2026-08-21：局势收尾即失效，
        ### 下次辩论重新打标）
        international_organization:th_gensokyo_io = {
            every_international_organization_member = {
                remove_variable = th_gensokyo_crusade_participant
                remove_variable = th_gensokyo_crusade_supporter
            }
        }
    }
}
