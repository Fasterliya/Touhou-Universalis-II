th_gensokyo_culture_chronicle = {
	monthly_spawn_chance = 1
	hint_tag = hint_th_gensokyo_culture_chronicle

	can_start = {
		always = yes
	}

	can_end = {
		always = no
	}

	visible = {
		is_human = yes
	}

	on_start = {
		every_country = {
			limit = { is_human = yes }
			set_variable = { name = th_cc_page value = 0 }
			set_variable = { name = th_cc_page_start value = 0 }
			set_variable = { name = th_cc_page_end value = 7 }
			th_culture_chronicle_refresh_lists = yes
		}
	}
}
