### 对决战终局判定：任一方分数达到隐藏终点
th_danmaku_duel_finished_trigger = {
    has_variable = th_danmaku_duel_goal_var
    OR = {
        var:th_danmaku_duel_player_score_var >= var:th_danmaku_duel_goal_var
        var:th_danmaku_duel_enemy_score_var >= var:th_danmaku_duel_goal_var
    }
}
