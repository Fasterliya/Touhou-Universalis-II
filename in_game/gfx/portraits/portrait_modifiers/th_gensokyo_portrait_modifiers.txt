th_gensokyo_portrait = {

	touhou_marisa = {	#这里是某个具体的头像
		dna_modifiers = {
			accessory = {
				mode = replace
				gene = th_gensokyo_portrait_group_1	#在这里和gene里写的对上
				template = touhou_marisa	#在这里和对应头像对上
				value = 0.5
			}
		}
		weight = {
			base = 0
			modifier = {
				# DONT change this unless you really need override particular character.
				add = 200

				# 通常使用特质标志纸片人，但你也可以开发自己的机制
				# 不过不要让一个角色身上有多个纸片人立绘
				th_portrait_trigger = yes
				#上述scripted_trigger中有对应ethnicity的内容，从而能造成对应基因组的控制
				#因为相同的ethnicity会母系传给女性，父系传给男性
				#所以新建的角色请一定让他们从ethnicity上算是纸片人
				
				#我自己之前忘记写的东西，具体的ethnicity判定，确保这个头像组的头像在一个遗传组里
				ethnicity ?= ethnicity:th_gensokyo_ethnicity

				var:portrait_locked ?= flag:touhou_marisa
				#上述的trigger_if是用于检测的，如果有锁定肖像的变量
				#那么这个变量应该是什么值才会让这个角色取到头像对应的角色
				#如果没有就无事发生
				#特别的，对于没做过乔米尼引擎mod的人：?=的意思就是检查是否等于这个值，也就是老版本的=中完全相等那部分职能
				#其实现在=也能用，不过?=会先行检测前方内容的存在性，而在不存在时不输出报错
				#不过仅限当不满足时会立刻停止检测的地方，如果并非——比如决议
				#诸如此类需要继续检测用以满足显示有哪些条件不满足的地方，则先行检测作用域是否存在并使用trigger_if和trigger_else更好
				#这十分有助于玩家的C盘减小负担，规范代码，从我做起
			}
		}
	}
