### ============================================================
### 贤者议会 — 开局人口
### 固定三贤者：迷途之家(T05)、后户之国(T13)、华扇的仙界(T83)
### 贤者威望初始值 100（设计文档：贤者威望开局即巅峰，随团结度自然波动）
### 成员scope预存到IO变量 (供GUI data function读取)
### ============================================================

international_organization_manager = {
    add_international_organization = {
        type = th_sages_council_io
        creation_date = 1337.4.1
        members = { T05 T13 T83 }
    }
}

# 初始化贤者威望 + 阈值应用 + 成员scope预存
th_sages_council_io = {

    set_variable = {
        name = th_sages_prestige_var
        value = 100
    }
    th_sages_council_apply_prestige_thresholds_effect = yes

    # 预存三位贤者国家scope引用
    # 中央 — 迷途之家 T05
    every_member_country = {
        limit = { this = c:T05 }
        save_scope_as = th_sage_center_scope
    }
    set_variable = {
        name = th_sage_center_ref
        value = scope:th_sage_center_scope
    }
    clear_saved_scope = th_sage_center_scope

    # 左侧 — 后户之国 T13
    every_member_country = {
        limit = { this = c:T13 }
        save_scope_as = th_sage_left_scope
    }
    set_variable = {
        name = th_sage_left_ref
        value = scope:th_sage_left_scope
    }
    clear_saved_scope = th_sage_left_scope

    # 右侧 — 华扇的仙界 T83
    every_member_country = {
        limit = { this = c:T83 }
        save_scope_as = th_sage_right_scope
    }
    set_variable = {
        name = th_sage_right_ref
        value = scope:th_sage_right_scope
    }
    clear_saved_scope = th_sage_right_scope
}
